package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryTranslationsTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testTranslationsDeserializationViaAlphaCode() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTranslationsDeserializationViaName() {
        given()
            .when()
                .get("/v1/name/France")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTranslationsDeserializationViaCurrency() {
        given()
            .when()
                .get("/v1/currency/USD")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTranslationsDeserializationViaCapital() {
        given()
            .when()
                .get("/v1/capital/London")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTranslationsDeserializationViaRegion() {
        given()
            .when()
                .get("/v1/region/Europe")
            .then()
                .statusCode(200);
    }
}
