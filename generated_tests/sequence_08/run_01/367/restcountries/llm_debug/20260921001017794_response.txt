package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.Iterator;
import java.util.List;
import java.util.Map;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class LanguageTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getProperty("baseURI");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    private Map getFirstLanguage(Response response) {
        Object languagesObj = response.jsonPath().get("languages");
        if (languagesObj instanceof List) {
            List list = (List) languagesObj;
            return (Map) list.get(0);
        } else if (languagesObj instanceof Map) {
            Map map = (Map) languagesObj;
            Iterator it = map.values().iterator();
            return (Map) it.next();
        } else {
            throw new AssertionError("languages structure not recognized");
        }
    }

    @Test(timeout = 60000)
    public void testSetIso639_1() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v1/alpha/US");
        response.then().statusCode(200);
        Map lang = getFirstLanguage(response);
        assertEquals("en", lang.get("iso639_1"));
    }

    @Test(timeout = 60000)
    public void testSetIso639_2() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v1/alpha/US");
        response.then().statusCode(200);
        Map lang = getFirstLanguage(response);
        assertEquals("eng", lang.get("iso639_2"));
    }

    @Test(timeout = 60000)
    public void testSetName() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v1/alpha/US");
        response.then().statusCode(200);
        Map lang = getFirstLanguage(response);
        assertEquals("English", lang.get("name"));
    }

    @Test(timeout = 60000)
    public void testSetNativeName() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v1/alpha/US");
        response.then().statusCode(200);
        Map lang = getFirstLanguage(response);
        assertEquals("English", lang.get("nativeName"));
    }
}
