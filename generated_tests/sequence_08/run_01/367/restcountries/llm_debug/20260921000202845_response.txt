package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CurrencyTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("test.base.url", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testCurrencySetCode() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testCurrencySetName() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testCurrencySetSymbol() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].symbol", equalTo("$"));
    }
}
