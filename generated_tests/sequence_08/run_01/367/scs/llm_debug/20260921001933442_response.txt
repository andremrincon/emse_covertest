package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    private final String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");

    @Test(timeout = 60000)
    public void testCostfuns_iEquals5() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/5/x")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iGreaterThan666() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/667/x")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_sEqualsBaab() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/0/baab")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_sCompareToEqualsZero() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/0/ababba")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iEqualsMinus4AndSEqualsAbab() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/-4/abab")
        .then()
            .statusCode(200)
            .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iLessThanMinus444() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/costfuns/-445/abab")
        .then()
            .statusCode(200)
            .body(equalTo("6"));
    }
}
