package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;

public class RegexTest {

    @Test(timeout = 60000)
    public void testUrlPatternMatch() {
        given()
            .baseUri("http://localhost:8080")
            .pathParam("txt", "http://abc/def")
            .when()
            .get("/api/pat/{txt}")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDatePatternMatch() {
        given()
            .baseUri("http://localhost:8080")
            .pathParam("txt", "mon01jan")
            .when()
            .get("/api/pat/{txt}")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFpePatternMatch() {
        given()
            .baseUri("http://localhost:8080")
            .pathParam("txt", "1.2e+34")
            .when()
            .get("/api/pat/{txt}")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNonePatternMatch() {
        given()
            .baseUri("http://localhost:8080")
            .pathParam("txt", "hello")
            .when()
            .get("/api/pat/{txt}")
            .then()
            .statusCode(200);
    }
}
