package ts01qwen3_7_plus;

import io.restassured.RestAssured;

public class CountryRestV1Test {

        RestAssured.baseURI = "http://localhost:8080/rest";

    @Test(timeout = 60000)
    public void testGetByAlphaSuccess() {
        given()
            .contentType(ContentType.JSON)
        .when()
            .get("/v1/alpha/US")
        .then()
            .statusCode(200);
