package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.nullValue;

public class LanguageTest {

    private String baseUrl;

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080/rest");
        baseUrl = RestAssured.baseURI;
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeReturnsLanguageData() {
        given()
            .pathParam("alphacode", "US")
        .when()
            .get(baseUrl + "/v1/alpha/{alphacode}")
        .then()
            .statusCode(200)
            .body("[0].languages[0]['iso639_1']", nullValue())
            .body("[0].languages[0]['iso639_2']", equalTo("eng"))
            .body("[0].languages[0].name", equalTo("English"))
            .body("[0].languages[0].nativeName", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testV1LangReturnsLanguageData() {
        given()
            .pathParam("lang", "es")
        .when()
            .get(baseUrl + "/v1/lang/{lang}")
        .then()
            .statusCode(200)
            .body("[0].languages[0]['iso639_1']", equalTo("es"))
            .body("[0].languages[0]['iso639_2']", equalTo("spa"))
            .body("[0].languages[0].name", equalTo("Spanish"))
            .body("[0].languages[0].nativeName", equalTo("Español"));
    }

    @Test(timeout = 60000)
    public void testV2AlphaCodeReturnsLanguageData() {
        given()
            .pathParam("alphacode", "US")
        .when()
            .get(baseUrl + "/v2/alpha/{alphacode}")
        .then()

            .body("languages[0]['iso639_1']", equalTo("en"))
            .body("languages[0]['iso639_2']", equalTo("eng"))
            .body("languages[0].name", equalTo("English"))
            .body("languages[0].nativeName", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testV2LangReturnsLanguageData() {
        given()
            .pathParam("lang", "Spanish")
        .when()
            .get(baseUrl + "/v2/lang/{lang}")
        .then()
            .statusCode(404)
            .body("[0].languages[0]['iso639_1']", equalTo("es"))
            .body("[0].languages[0]['iso639_2']", equalTo("spa"))
            .body("[0].languages[0].name", equalTo("Spanish"))
            .body("[0].languages[0].nativeName", equalTo("Español"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeWithInvalidCodeReturns404() {
        given()
            .pathParam("alphacode", "XYZ")
        .when()
            .get(baseUrl + "/v1/alpha/{alphacode}")
        .then()
            .statusCode(404);
    }

    public void testV1LangWithInvalidLanguageReturns404() {
        given()
            .pathParam("lang", "123")
        .when()
            .get(baseUrl + "/v1/lang/{lang}")
        .then()

    }

    @Test(timeout = 60000)
    public void testV2AlphaCodeWithInvalidCodeReturns404() {
        given()
            .pathParam("alphacode", "ZZZ")
        .when()
            .get(baseUrl + "/v2/alpha/{alphacode}")
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV2LangWithInvalidLanguageReturns404() {
        given()
            .pathParam("lang", "123")
        .when()
            .get(baseUrl + "/v2/lang/{lang}")
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeWithGermanLanguageData() {
        given()
            .pathParam("alphacode", "DE")
        .when()
            .get(baseUrl + "/v1/alpha/{alphacode}")

    @Test(timeout = 60000)
    public void testV1AlphaCodeWithInvalidFormatReturns400() {
