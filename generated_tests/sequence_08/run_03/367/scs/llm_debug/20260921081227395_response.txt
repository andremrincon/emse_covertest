package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class PatTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testPatLenLessThanOrEqualTo2() {
        given().
        when().
            get("/api/pat/abc/ab").
        then().
            body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPatFoundNoPatRev() {
        given().
        when().
            get("/api/pat/abcde/abc").
        then().
            body(equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testPatAndPatRevImmediate() {
        given().
        when().
            get("/api/pat/abccba/abc").
        then().
            body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPatAndPatRevLater() {
        given().
        when().
            get("/api/pat/abcxxcba/abc").
        then().
            body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPatRevFoundNoPat() {
        given().
        when().
            get("/api/pat/cbaxxx/abc").
        then().
            body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testPatRevAndPatImmediate() {
        given().
        when().
            get("/api/pat/cbaabc/abc").
        then().
            body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPatRevAndPatLater() {
        given().
        when().
            get("/api/pat/cbaxxxabc/abc").
        then().
            body(equalTo("0"));
    }
}
