package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CostfunsTest {

    static {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null) {
            baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCostfunsIEquals5() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/5/x").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsIGreaterThan666() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/667/x").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsINotEqualsMinus4AndSNotEqualsAbab() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/0/x").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsSEqualsBaab() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/0/baab").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsSCompareToGreaterThan0() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/0/ababc").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsSCompareToGreaterThanOrEquals0() {
        given().when().get("/api/costfuns/1/a").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/0/ababba").then().statusCode(200);
    }
}
