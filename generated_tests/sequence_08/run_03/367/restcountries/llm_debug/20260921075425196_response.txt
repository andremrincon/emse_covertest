package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class LanguageTest {

    @Before
    public void setUp() {
        String baseUri = System.getenv("BASE_URI");
        if (baseUri == null || baseUri.isEmpty()) {
            baseUri = System.getProperty("baseURI", "http://localhost:8080/rest");
        }
        RestAssured.baseURI = baseUri;
    }

    @Test(timeout = 60000)
    public void testSetIso639_1ViaV1Alpha() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSetIso639_2ViaV1Lang() {
        given()
            .when()
                .get("/v1/lang/es")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSetNameViaV2Alpha() {
        given()
            .when()
                .get("/v2/alpha/US")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSetNativeNameViaV2Lang() {
        given()
            .when()
                .get("/v2/lang/Spanish")
            .then()
                .statusCode(200);
    }
}
