package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class LanguageTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080/rest";
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testSetIso639_1() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("languages[0].iso639_1", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void testSetIso639_2() {
        given()
            .when()
                .get("/v2/alpha/US?fields=languages")
            .then()
                .body("languages[0].iso639_2", equalTo("eng"));
    }

    @Test(timeout = 60000)
    public void testSetName() {
        given()
            .when()
                .get("/v1/lang/es")
            .then()
                .body("[0].languages[0].name", equalTo("Spanish"));
    }

    @Test(timeout = 60000)
    public void testSetNativeName() {
        given()
            .when()
                .get("/v2/lang/Spanish?fields=languages")
            .then()
                .body("[0].languages[0].nativeName", equalTo("Español"));
    }
}
