        RestAssured.baseURI = BASE_URL;

    @Test(timeout = 60000)
    public void testGetByAlphaValidCodeWithFields() {
        given()
            .queryParam("fields", "name;capital;population")
        .when()
            .get("/v2/alpha/US")
        .then()
            .statusCode(200)
