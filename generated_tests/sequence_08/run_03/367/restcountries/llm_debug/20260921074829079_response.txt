package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.notNullValue;

public class CountryTranslationsTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseURI", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeUS_Status200() {
        given()
            .pathParam("alphacode", "US")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeUS_TranslationsDeNotNull() {
        given()
            .pathParam("alphacode", "US")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .body("translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetAlphaCode123_Status400() {
        given()
            .pathParam("alphacode", "123")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeXYZ_Status404() {
        given()
            .pathParam("alphacode", "XYZ")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeTrue_Status404() {
        given()
            .pathParam("alphacode", "True")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(404);
    }
}
