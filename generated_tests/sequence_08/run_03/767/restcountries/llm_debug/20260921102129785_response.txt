package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryServiceBaseTest {

    private final String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");

    @Test(timeout = 60000)
    public void testLoadJson() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/v1/all")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha2Char() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("alphacode", "US")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha3Char() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("alphacode", "USA")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCodeList() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .queryParam("codes", "US,CA")
        .when()
            .get("/v1/alpha")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearchName() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("name", "France")
            .queryParam("fullText", true)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearchAltSpelling() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("name", "French%20Republic")
            .queryParam("fullText", true)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubstringSearchName() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("name", "Fran")
            .queryParam("fullText", false)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubstringSearchAltSpelling() {
        given().baseUri(baseUrl).when().get("/v1/all").then().statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
            .pathParam("name", "French%20Rep")
            .queryParam("fullText", false)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }
}
