package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CurrencyTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("test.base.uri", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEUR() {
        given()
            .when()
            .get("/v2/currency/EUR")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1CurrencyUSD() {
        given()
            .when()
            .get("/v1/currency/USD")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV2AlphaUS() {
        given()
            .when()
            .get("/v2/alpha/US")
            .then()
            .statusCode(200);
    }
}
