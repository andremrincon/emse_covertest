package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CORSFilterTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080";
        RestAssured.basePath = "/rest";
    }

    @Test(timeout = 60000)
    public void testCorsFilterAddsAllowOriginHeader() {
        given()
        .when()
            .get("/v1/all")
        .then()
            .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testCorsFilterAddsAllowMethodsHeader() {
        given()
        .when()
            .get("/v1/alpha/US")
        .then()
            .header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testCorsFilterAddsAllowHeadersHeader() {
        given()
        .when()
            .get("/v1/currency/USD")
        .then()
            .header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void testCorsFilterAddsCacheControlHeader() {
        given()
        .when()
            .get("/v1/name/France")
        .then()
            .header("Cache-Control", "public, max-age=86400");
    }
