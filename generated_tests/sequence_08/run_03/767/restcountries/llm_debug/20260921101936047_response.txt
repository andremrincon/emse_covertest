package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class ContributionTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testContributeWithValidPayload() {
        given()
            .contentType("application/json")
            .body("{\"amount\": 100, \"token\": \"tok_12345\"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(202);
    }

    @Test(timeout = 60000)
    public void testContributeWithInvalidAmountType() {
        given()
            .contentType("application/json")
            .body("{\"amount\": \"invalid\", \"token\": \"tok_12345\"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeWithMissingToken() {
        given()
            .contentType("application/json")
            .body("{\"amount\": 50}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeWithEmptyJson() {
        given()
            .contentType("application/json")
            .body("{}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeWithMalformedJson() {
        given()
            .contentType("application/json")
            .body("{malformed}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }
}
