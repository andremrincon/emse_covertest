package ts01qwen3_7_plus_2;

import org.junit.Test;

import static io.restassured.RestAssured.given;

public class ExpintTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testExpintXGreaterThan1() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/3/2.5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintXLessThanOrEqualTo1() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/3/0.1")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintNLessThan0() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/-1/2.5")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpintXLessThan0() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/3/-999.9")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpintNEquals0() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/0/2.5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintXEquals0NGreaterThan1() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/expint/3/0.0")
        .then()
            .statusCode(200);
    }
}
