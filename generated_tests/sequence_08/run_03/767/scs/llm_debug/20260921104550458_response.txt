import org.junit.Test;
import static io.restassured.RestAssured.given;

public class RegexTest {

    @Test(timeout = 60000)
    public void testSubjectReturnsUrl() {
        given()
            .pathParam("txt", "http://a/b")
        .when()
            .get("http://localhost:8080/api/pat/{txt}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)

    }
