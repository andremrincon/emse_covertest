import org.junit.Test;

    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080";

    @Test(timeout = 60000)
    public void testBessj_nLessThan2_returns400() {
        given()
            .pathParam("n", 1)
            .pathParam("x", 2.5)

            .get("/api/bessj/{n}/{x}")
        .then()
            .statusCode(400);
