package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;

public class CountryRestV1Test {

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testSetCurrencyCodeNameSymbolViaV2Alpha() {
        given()
        .when()
            .get("/v2/alpha/US")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSetCurrencyCodeNameSymbolViaV2Currency() {
        given()
        .when()
            .get("/v2/currency/EUR")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSetCurrencyCodeNameSymbolViaV1Currency() {
        given()
        .when()
            .get("/v1/currency/USD")
        .then()
            .statusCode(200);
    }
}
