package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.containsString;
import static org.hamcrest.Matchers.lessThan;

public class LanguageTest {

    @Test(timeout = 60000)
    public void testSetIso639_1() {
        given()
            .when()
                .get("http://localhost:8080/rest/v1/alpha/US")
            .then()
                .statusCode(lessThan(300))
                .body(containsString("\"iso639_1\":\"en\""));
    }

    @Test(timeout = 60000)
    public void testSetIso639_2() {
        given()
            .when()
                .get("http://localhost:8080/rest/v1/alpha/US")
            .then()
                .statusCode(lessThan(300))
                .body(containsString("\"iso639_2\":\"eng\""));
    }

    @Test(timeout = 60000)
    public void testSetName() {
        given()
            .when()
                .get("http://localhost:8080/rest/v1/alpha/US")
            .then()
                .statusCode(lessThan(300))
                .body(containsString("\"name\":\"English\""));
    }

    @Test(timeout = 60000)
    public void testSetNativeName() {
        given()
            .when()
                .get("http://localhost:8080/rest/v1/alpha/US")
            .then()
                .statusCode(lessThan(300))
                .body(containsString("\"nativeName\":\"English\""));
    }
}
