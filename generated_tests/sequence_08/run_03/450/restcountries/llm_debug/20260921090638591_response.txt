package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryRestV2Test {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testGetByAlphaSuccessWithFields() {
        Response response = given()
            .queryParam("fields", "name")
        .when()
            .get("/v2/alpha/US");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaListSuccessWithFields() {
        Response response = given()
            .queryParam("codes", "US,CA")
            .queryParam("fields", "name")
        .when()
            .get("/v2/alpha");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaListNotFound() {
        Response response = given()
            .queryParam("codes", "XX,YY,ZZ")

            .get("/v2/alpha");
        response.then().statusCode(404);
    }
