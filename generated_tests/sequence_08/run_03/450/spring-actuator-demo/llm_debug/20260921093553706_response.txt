package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class SampleControllerTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testSayHelloWithDefaultName() {
        given()
        .when()
            .get("/")
        .then()
            .body(equalTo("Hello Guest!!"));
    }

    @Test(timeout = 60000)
    public void testSayHelloWithCustomName() {
        given()
        .when()
            .get("/?name=John Smith")
        .then()
            .body(equalTo("Hello John Smith!!"));
    }

    @Test(timeout = 60000)
    public void testSlowApiWithZeroDelay() {
        given()
        .when()
            .get("/slowApi?delay=0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiWithPositiveDelay() {
        given()
        .when()
            .get("/slowApi?delay=1")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiWithNegativeDelay() {
        given()
        .when()
            .get("/slowApi?delay=-1")
        .then()
            .statusCode(500);
    }
}
