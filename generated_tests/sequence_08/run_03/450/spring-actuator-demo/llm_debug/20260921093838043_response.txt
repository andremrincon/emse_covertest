package ts01qwen3_7_plus_2;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class SampleControllerTest {

    @Test(timeout = 60000)
    public void testSayHelloDefaultName() {
        given()
                .baseUri("http://localhost:8080")
                .when()
                .get("/")
                .then()
                .statusCode(200)
                .body(equalTo("Hello Guest!!"));
    }

    @Test(timeout = 60000)
    public void testSayHelloCustomName() {
        given()
                .baseUri("http://localhost:8080")
                .queryParam("name", "John Smith")
                .when()
                .get("/")
                .then()
                .statusCode(200)
                .body(equalTo("Hello John Smith!!"));
    }

    @Test(timeout = 60000)
    public void testSlowApiZeroDelay() {
        given()
                .baseUri("http://localhost:8080")
                .queryParam("delay", 0)
                .when()
                .get("/slowApi")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiPositiveDelay() {
        given()
                .baseUri("http://localhost:8080")
                .queryParam("delay", 1)
                .when()
                .get("/slowApi")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiNegativeDelay() {
        given()
                .baseUri("http://localhost:8080")
                .queryParam("delay", -1)
                .when()
                .get("/slowApi")
                .then()
                .statusCode(500);
    }
}
