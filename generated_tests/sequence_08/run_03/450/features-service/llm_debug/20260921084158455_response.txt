package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ProductConfigurationTest {

    @BeforeClass
    public static void setupClass() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testActivateFeature() {
        String pid = "P-" + UUID.randomUUID().toString();
        String fid = "F-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + fid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));

        given()
            .when()
                .post("/products/" + pid + "/configurations/" + cid + "/features/" + fid)
            .then()
                .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetActiveFeatures() {
        String pid = "P-" + UUID.randomUUID().toString();
        String fid = "F-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + fid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid + "/features/" + fid).then().statusCode(lessThan(300));

        given()
            .when()
                .get("/products/" + pid + "/configurations/" + cid + "/features")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDeactivateFeature() {
        String pid = "P-" + UUID.randomUUID().toString();
        String fid = "F-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + fid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid + "/features/" + fid).then().statusCode(lessThan(300));

        given()
            .when()
                .delete("/products/" + pid + "/configurations/" + cid + "/features/" + fid)
            .then()
                .statusCode(204);
    }

    @Test(timeout = 60000)
    public void testGetConfiguration() {
        String pid = "P-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));

        given()
            .when()
                .get("/products/" + pid + "/configurations/" + cid)
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetProduct() {
        String pid = "P-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));

        given()
            .when()
                .get("/products/" + pid)
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAddRequiresConstraint() {
        String pid = "P-" + UUID.randomUUID().toString();
        String f1 = "F1-" + UUID.randomUUID().toString();
        String f2 = "F2-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f1).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f2).then().statusCode(lessThan(300));

        given()
            .contentType(ContentType.URLENC)
            .formParam("sourceFeature", f1)
            .formParam("requiredFeature", f2)
            .when()
                .post("/products/" + pid + "/constraints/requires")
            .then()
                .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testActivateFeatureWithConstraint() {
        String pid = "P-" + UUID.randomUUID().toString();
        String f1 = "F1-" + UUID.randomUUID().toString();
        String f2 = "F2-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f1).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f2).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));
        given()
            .contentType(ContentType.URLENC)
            .formParam("sourceFeature", f1)
            .formParam("requiredFeature", f2)
            .when()
                .post("/products/" + pid + "/constraints/requires")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .post("/products/" + pid + "/configurations/" + cid + "/features/" + f1)
            .then()
                .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetConfigurationAfterConstraint() {
        String pid = "P-" + UUID.randomUUID().toString();
        String f1 = "F1-" + UUID.randomUUID().toString();
        String f2 = "F2-" + UUID.randomUUID().toString();
        String cid = "C-" + UUID.randomUUID().toString();

        given().when().post("/products/" + pid).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f1).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/features/" + f2).then().statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid).then().statusCode(lessThan(300));
        given()
            .contentType(ContentType.URLENC)
            .formParam("sourceFeature", f1)
            .formParam("requiredFeature", f2)
            .when()
                .post("/products/" + pid + "/constraints/requires")
            .then()
                .statusCode(lessThan(300));
        given().when().post("/products/" + pid + "/configurations/" + cid + "/features/" + f1).then().statusCode(lessThan(300));

        given()
            .when()
                .get("/products/" + pid + "/configurations/" + cid)
            .then()
                .statusCode(200);
    }
}
