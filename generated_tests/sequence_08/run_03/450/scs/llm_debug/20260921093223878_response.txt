package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CalcTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCalcPlusOperation() {
        given().when().get("/api/calc/plus/1.0/1.0").then().statusCode(lessThan(300));

        String response = given().when().get("/api/calc/plus/15.5/4.5").then().extract().asString();

        assertEquals("20.0", response);
    }

    @Test(timeout = 60000)
    public void testCalcPiConstant() {
        given().when().get("/api/calc/pi/0.0/0.0").then().statusCode(lessThan(300));

        String response = given().when().get("/api/calc/pi/0.0/0.0").then().extract().asString();

        assertEquals("3.141592653589793", response);
    }
}
