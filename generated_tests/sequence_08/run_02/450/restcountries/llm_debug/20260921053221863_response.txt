package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CORSFilterTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        RestAssured.baseURI = baseUrl != null ? baseUrl : "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testDeTranslation() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("translations.de", equalTo("Vereinigte Staaten von Amerika"));
    }

    @Test(timeout = 60000)
    public void testEsTranslation() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("translations.es", equalTo("Estados Unidos"));
    }

    @Test(timeout = 60000)
    public void testFrTranslation() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("translations.fr", equalTo("États-Unis"));
    }

    @Test(timeout = 60000)
    public void testJaTranslation() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("translations.ja", equalTo("アメリカ合衆国"));
    }

    @Test(timeout = 60000)
    public void testItTranslation() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("translations.it", equalTo("Stati Uniti D'America"));
    }
}
