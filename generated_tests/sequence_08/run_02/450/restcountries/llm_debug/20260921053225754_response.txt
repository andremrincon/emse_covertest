package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CORSFilterTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_BadRequest() {
        given()
                .when()
                .get("/v2/alpha/A")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_Success() {
        given()
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_BadRequest() {
        given()
                .when()
                .get("/v2/alpha?codes=")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_Success() {
        given()
                .when()
                .get("/v2/alpha?codes=US,CA")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_Exception() {
        given()
                .when()
                .get("/v2/alpha?codes=[\"US\", \"CA\"]")
                .then()
                .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_BadRequest() {

                .then()
                .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testGetByName_Success() {

                .when()

    }

    @Test(timeout = 60000)
    public void testGetByCallingCode_Success() {
        given()
                .when()

                .then()
                .statusCode(200);

    @Test(timeout = 60000)

                .statusCode(200);
