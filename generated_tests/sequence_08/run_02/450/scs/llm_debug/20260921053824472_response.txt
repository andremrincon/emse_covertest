package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;

public class CalcTest {

    @Test(timeout = 60000)
    public void testPiOperator() {
        given().when().get("http://localhost:8080/api/calc/pi/0/0").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEOperator() {
        given().when().get("http://localhost:8080/api/calc/e/0/0").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnaryOperators() {
        String[] ops = {"sqrt", "log", "sine", "cosine", "tangent"};
        for (String op : ops) {
            given().when().get("http://localhost:8080/api/calc/" + op + "/10/0").then().statusCode(200);
        }
    }

    @Test(timeout = 60000)
    public void testBinaryOperators() {
        String[] ops = {"plus", "subtract", "multiply", "divide"};
        for (String op : ops) {
            given().when().get("http://localhost:8080/api/calc/" + op + "/10/2").then().statusCode(200);
        }
    }

    @Test(timeout = 60000)
    public void testDivideByZero() {
        given().when().get("http://localhost:8080/api/calc/divide/10/0").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testInvalidNumberFormat() {
        given().when().get("http://localhost:8080/api/calc/plus/abc/2").then().statusCode(400);
    }
}
