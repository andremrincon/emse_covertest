package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class NotyPevarTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testNotyPevar_i0True_i3True() {
        Response response = given()
            .pathParam("i", 28)
            .pathParam("s", "example-string")
        .when()
            .get("/api/notypevar/{i}/{s}");

        response.then().body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevar_i1True_i2True_i3True() {
        Response response = given()
            .pathParam("i", 7)
            .pathParam("s", "zebra")
        .when()
            .get("/api/notypevar/{i}/{s}");

        response.then().body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevar_i0False_i1False_i2False_i3False() {
        Response response = given()
            .pathParam("i", 0)
            .pathParam("s", "apple")
        .when()
            .get("/api/notypevar/{i}/{s}");

        response.then().body(equalTo("0"));
    }
}
