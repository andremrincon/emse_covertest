package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class RemainderTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveAPositiveB() {
        given()
            .when()
                .get("/api/remainder/17/5")
            .then()
                .statusCode(200)
                .body("result", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveANegativeB() {
        given()
            .when()
                .get("/api/remainder/17/-5")
            .then()
                .statusCode(200)
                .body("result", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeAPositiveB() {
        given()
            .when()
                .get("/api/remainder/-17/5")
            .then()
                .statusCode(200)
                .body("result", equalTo(-2));
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeANegativeB() {
        given()
            .when()
                .get("/api/remainder/-17/-5")
            .then()
                .statusCode(200)
                .body("result", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testRemainderZeroA() {
        given()
            .when()
                .get("/api/remainder/0/5")
            .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderZeroB() {
        given()
            .when()
                .get("/api/remainder/5/0")
            .then()
                .statusCode(400);
    }
}
