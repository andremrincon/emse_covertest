package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class GammqTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testExpintN0XPositive() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/0/2.5");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintN2XZero() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/2/0.0");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintN3XGreaterThan1() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/3/2.5");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintN1XLessThan1() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/1/0.5");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintN3XLessThan1() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/3/0.5");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintNegativeN() {
        Response response = given()
            .baseUri(BASE_URL)
            .when()
            .get("/api/expint/-1/2.5");
        response.then().statusCode(400);
    }
}
