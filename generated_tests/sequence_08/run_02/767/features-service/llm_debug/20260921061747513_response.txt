package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.not;

public class ConstraintRequiresTest {

    private static String baseUri;

    @BeforeClass
    public static void setup() {
        baseUri = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        RestAssured.baseURI = baseUri;
    }

    @Test(timeout = 60000)
    public void testCreateRequiresConstraint() {
        String productName = "Enterprise-Server-X1-" + UUID.randomUUID().toString();
        String sourceFeature = "RAID-Controller-Card";
        String requiredFeature = "128GB-ECC-RAM";

        given().baseUri(baseUri).when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));

        given().baseUri(baseUri)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/" + productName + "/constraints/requires")
                .then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfiguration_SourceActive_RequiredInactive() {
        String productName = "Desktop-PC-" + UUID.randomUUID().toString();
        String sourceFeature = "SSD-Storage";
        String requiredFeature = "Graphics-Card";
        String configName = "Config-2";

        given().baseUri(baseUri).when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName).then().statusCode(lessThan(300));
        given().baseUri(baseUri)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName + "/features/" + sourceFeature).then().statusCode(lessThan(300));

        given().baseUri(baseUri).when().get("/products/" + productName + "/configurations/" + configName + "/features")
                .then().statusCode(200).body("$", hasItem(requiredFeature));
    }

    @Test(timeout = 60000)
    public void testEvaluateConfiguration_SourceActive_RequiredActive() {
        String productName = "Workstation-Z8-" + UUID.randomUUID().toString();
        String sourceFeature = "CPU-i9-13900H";
        String requiredFeature = "Integrated-Graphics-Only";
        String configName = "Config-3";

        given().baseUri(baseUri).when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName).then().statusCode(lessThan(300));
        given().baseUri(baseUri)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName + "/features/" + requiredFeature).then().statusCode(lessThan(300));

        given().baseUri(baseUri).when().get("/products/" + productName + "/configurations/" + configName + "/features")
                .then().statusCode(200).body("$", hasItem(requiredFeature));
    }

    @Test(timeout = 60000)
    public void testEvaluateConfiguration_SourceInactive() {
        String productName = "Smart-Watch-S3-" + UUID.randomUUID().toString();
        String sourceFeature = "Strap-Leather";
        String requiredFeature = "OS-Home";
        String configName = "Config-4";

        given().baseUri(baseUri).when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName).then().statusCode(lessThan(300));
        given().baseUri(baseUri)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().baseUri(baseUri).when().post("/products/" + productName + "/configurations/" + configName + "/features/" + requiredFeature).then().statusCode(lessThan(300));

        given().baseUri(baseUri).when().get("/products/" + productName + "/configurations/" + configName + "/features")
                .then().statusCode(200).body("$", not(hasItem(sourceFeature)));
    }

    @Test(timeout = 60000)
    public void testGetProductWithRequiresConstraint() {
        String productName = "Mobile-App-Basic-" + UUID.randomUUID().toString();
        String sourceFeature = "Push-Notifications";
        String requiredFeature = "User-Authentication";

        given().baseUri(baseUri).when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri).formParam("description", "desc").when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));
        given().baseUri(baseUri)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));

        given().baseUri(baseUri).when().get("/products/" + productName)
                .then().statusCode(200).body("constraints[0].type", equalTo("requires"));
    }
}
