package ts01qwen3_7_plus;

    @Test(timeout = 60000)
    public void bessj_n_less_than_2_returns_400() {
        given()
            .pathParam("n", 2)
            .pathParam("x", 2.5)
        .when()
            .get("/api/bessj/{n}/{x}")
        .then()
            .statusCode(400);
