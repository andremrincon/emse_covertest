package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class NcsRestTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testFisherCatchBlockWithInvalidX() {
        Response response = given()
                .pathParam("m", 10)
                .pathParam("n", 5)
                .pathParam("x", 1.2)
                .when()
                .get("/api/fisher/{m}/{n}/{x}");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testFisherCatchBlockWithZeroX() {
        Response response = given()
                .pathParam("m", 10)
                .pathParam("n", 5)
                .pathParam("x", 0.0)
                .when()
                .get("/api/fisher/{m}/{n}/{x}");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderSuccessExecution() {
        Response response = given()
                .pathParam("a", 17)
                .pathParam("b", 5)
                .when()
                .get("/api/remainder/{a}/{b}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderWithNegativeValues() {
        Response response = given()
                .pathParam("a", -9)
                .pathParam("b", 4)
                .when()
                .get("/api/remainder/{a}/{b}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderWithZeroA() {
        Response response = given()
                .pathParam("a", 0)
                .pathParam("b", 5)
                .when()
                .get("/api/remainder/{a}/{b}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherSuccessExecution() {
        Response response = given()
                .pathParam("m", 10)
                .pathParam("n", 5)
                .pathParam("x", 0.75)
                .when()
                .get("/api/fisher/{m}/{n}/{x}");

        response.then().statusCode(200);
    }
}
