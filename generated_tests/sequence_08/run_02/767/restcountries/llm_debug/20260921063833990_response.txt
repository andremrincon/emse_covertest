package ts01qwen3_7_plus_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CORSFilterTest {

    private static final String BASE_URI = System.getenv("BASE_URI") != null ? System.getenv("BASE_URI") : "http://localhost:8080/rest";

    @Before
    public void setUp() {
        RestAssured.baseURI = BASE_URI;
    }

    @Test(timeout = 60000)
    public void testCorsAllowOriginOnV1All() {
        given()
            .baseUri(BASE_URI)
        .when()
            .get("/v1/all")
        .then()
            .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowMethodsOnV2All() {
        given()
            .baseUri(BASE_URI)
        .when()
            .get("/v2/all")
        .then()
            .header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowHeadersOnV1Alpha() {
        given()
            .baseUri(BASE_URI)
        .when()
            .get("/v1/alpha/US")
