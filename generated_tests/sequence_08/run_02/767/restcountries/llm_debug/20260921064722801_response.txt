import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryRestV1Test {

    private String baseUrl;

    @Before
    public void setUp() {
        baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080/rest";
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGetByAlphaValid() {
        Response response = given()
            .pathParam("alphacode", "US")
        .when()
            .get("/v1/alpha/{alphacode}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaInvalid() {
        Response response = given()
            .pathParam("alphacode", "123")
        .when()
            .get("/v1/alpha/{alphacode}");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaNotFound() {
        Response response = given()
            .pathParam("alphacode", "XYZ")

        response.then().statusCode(404);
    }
