package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class NotFoundExceptionMapperTest {

    private String getBaseUrl() {
        return System.getProperty("base.url", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testNotFoundExceptionMapperWithInvalidAlphaCode() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/v1/all")
        .then()
            .statusCode(lessThan(300));

        Response response = given()
            .baseUri(getBaseUrl())
        .when()
            .get("/v1/alpha/XYZ");

        response.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testNotFoundExceptionMapperWithInvalidName() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/v1/all")
        .then()
            .statusCode(lessThan(300));

        Response response = given()
            .baseUri(getBaseUrl())
        .when()
            .get("/v1/name/NonExistentCountry999");

        response.then().statusCode(404);
    }
}
