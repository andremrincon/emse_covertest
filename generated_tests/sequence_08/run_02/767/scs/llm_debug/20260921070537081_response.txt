package ts01qwen3_7_plus_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class CostfunsTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseURI", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testCostfuns_i5_sBaab() {
        given().when().get("/api/costfuns/5/baab").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/5/baab").then().statusCode(200).body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iNeg500_sAbab() {
        given().when().get("/api/costfuns/-500/abab").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/-500/abab").then().statusCode(200).body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_i700_sAbabba() {
        given().when().get("/api/costfuns/700/ababba").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/700/ababba").then().statusCode(200).body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iNeg4_sAbab() {
        given().when().get("/api/costfuns/-4/abab").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/-4/abab").then().statusCode(200).body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_i0_sAbabbb() {
        given().when().get("/api/costfuns/0/ababbb").then().statusCode(lessThan(300));
        given().when().get("/api/costfuns/0/ababbb").then().statusCode(200).body(equalTo("10"));
    }
}
