package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class LanguageTest {

    @Test(timeout = 60000)
    public void testV1AlphaCodeSetsLanguageFields() {
        // Arrange
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));

        // Act
        Response response = given().when().get("/v1/alpha/US");

        // Assert
        response.then().statusCode(200);
        assertEquals("en", response.jsonPath().getString("languages[0].iso639_1"));
        assertEquals("eng", response.jsonPath().getString("languages[0].iso639_2"));
        assertEquals("English", response.jsonPath().getString("languages[0].name"));
        assertEquals("English", response.jsonPath().getString("languages[0].nativeName"));
    }

    @Test(timeout = 60000)
