package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    private String getBaseUrl() {
        return System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testBranch_iEquals5_and_sGreaterThanAbabba() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/5/test")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testBranch_iLessThanMinus444_and_sEqualsAbab() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/-500/abab")
        .then()
            .statusCode(200)
            .body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void testBranch_iGreaterThan666_and_sEqualsAbabba() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/700/ababba")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testBranch_iNotEqualsMinus4_and_sEqualsBaab() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/-4/baab")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testBranch_sLessThanAbabba() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/0/a")
        .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testInvalidIntegerParameter() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/one/test")
        .then()
            .statusCode(400);
    }
}
