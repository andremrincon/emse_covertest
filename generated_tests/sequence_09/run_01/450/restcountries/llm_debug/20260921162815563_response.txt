package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class LanguageTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("restAssured.baseUri");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testIso6391_from_v1_alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0]['iso639_1']", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void testIso6392_from_v1_alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0]['iso639_2']", equalTo("eng"));
    }

    @Test(timeout = 60000)
    public void testName_from_v1_alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0]['name']", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testNativeName_from_v1_alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0]['nativeName']", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testIso6391_from_v1_name_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].languages[0]['iso639_1']", equalTo("fr"));
    }

    @Test(timeout = 60000)
    public void testCurrency_v1_USD_languages_iso6391() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/currency/USD");
        resp.then().body("[0].languages[0]['iso639_1']", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void test_v1_alpha_invalid_numeric_returns_400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/123");
        resp.then().statusCode(404);
    }
}
