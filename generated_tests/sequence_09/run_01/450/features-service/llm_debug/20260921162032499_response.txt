package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class WrongProductConfigurationExceptionTest {

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = System.getProperty("API_BASE", System.getenv().getOrDefault("API_BASE", "http://localhost:8080"));
    }

    @Test(timeout = 60000)
    public void createConfiguration_withExcessivelyLongName_returnsInternalServerError() {
        String product = "prod-" + UUID.randomUUID().toString();
        String longConfigName = "a-very-long-configuration-name-that-exceeds-the-maximum-allowed-length-and-could-potentially-cause-a-buffer-overflow-or-database-truncation-error-leading-to-an-internal-server-error-because-the-system-was-not-designed-to-handle-such-long-inputs-gracefully";
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response resp = given().when().post("/products/{productName}/configurations/{configurationName}", product, longConfigName);
        assertEquals(500, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void addFeatureToConfiguration_afterSetup_returnsCreated() {
        String product = "prod-" + UUID.randomUUID().toString();
        String configuration = "standard-gpu-cluster-" + UUID.randomUUID().toString();
        String feature = "distributed-training-" + UUID.randomUUID().toString();
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response resp = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature);
        assertEquals(201, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void getConfiguration_withNumericName_returnsServerError() {
        String product = "prod-" + UUID.randomUUID().toString();
        String problematicConfigurationName = "12345";
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response resp = given().when().get("/products/{productName}/configurations/{configurationName}", product, problematicConfigurationName);
        assertEquals(500, resp.getStatusCode());
    }
}
