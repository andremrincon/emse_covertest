package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CalcTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null) base = System.getenv("API_BASE");
        if (base == null) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testCalcPiReturnsPiValue() {
        given().when().get("/api/pat/The").then().statusCode(lessThan(300));
        given().when().get("/api/calc/pi/0/0").then().body(equalTo(Double.toString(Math.PI)));
    }

    @Test(timeout = 60000)
    public void testCalcSqrtComputesSquareRoot() {
        given().when().get("/api/pat/quick").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sqrt/9/0").then().body(equalTo("3.0"));
    }

    @Test(timeout = 60000)
    public void testCalcSineComputesSine() {
        given().when().get("/api/pat/brown").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sine/0/0").then().body(equalTo("0.0"));
    }

    @Test(timeout = 60000)
    public void testCalcPlusAddsNumbers() {
        given().when().get("/api/pat/fox").then().statusCode(lessThan(300));
        given().when().get("/api/calc/plus/15.5/4.5").then().body(equalTo("20.0"));
    }

    @Test(timeout = 60000)
    public void testCalcDivideByZeroReturnsServerError() {
        given().when().get("/api/pat/jumps").then().statusCode(lessThan(300));
        given().when().get("/api/calc/divide/100/0").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testCalcUnknownOperatorReturnsZero() {
        given().when().get("/api/pat/lazy").then().statusCode(lessThan(300));
        given().when().get("/api/calc/unknown/1/1").then().body(equalTo("0.0"));
    }
}
