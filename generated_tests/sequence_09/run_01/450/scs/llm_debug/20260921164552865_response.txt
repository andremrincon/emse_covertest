package ts01gpt_5_mini_2;

import org.junit.Test;
import org.junit.BeforeClass;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class TitleTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testMaleDrReturns200() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response res = given().when().get("/api/title/male/dr");
        assertEquals(200, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testFemaleDrReturns200() {
        given().when().get("/api/pat/ABABCABAB/ABAB").then().statusCode(lessThan(300));
        Response res = given().when().get("/api/title/female/dr");
        assertEquals(200, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testNoneProfReturns200() {
        given().when().get("/api/text2txt/The/quick/brown").then().statusCode(lessThan(300));
        Response res = given().when().get("/api/title/none/prof");
        assertEquals(200, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testNeuterJonesReturns500() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response res = given().when().get("/api/title/neuter/Jones");
        assertEquals(500, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testFemaleJonesReturns500() {
        given().when().get("/api/calc/add/15.5/4.5").then().statusCode(lessThan(300));
        Response res = given().when().get("/api/title/female/Jones");
        assertEquals(500, res.getStatusCode());
    }
}
