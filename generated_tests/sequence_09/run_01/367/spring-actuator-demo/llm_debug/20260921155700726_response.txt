package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class SampleControllerTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultReturnsGuestGreeting() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().when().get("/");
        assertEquals("Hello Guest!!", resp.asString());
    }

    @Test(timeout = 60000)
    public void testSayHelloWithSpecialNameReturnsProperGreeting() {
        given().when().get("/").then().statusCode(lessThan(300));
        String name = "María-José O'Connor-Smith III";
        Response resp = given().queryParam("name", name).when().get("/");
        assertEquals("Hello María-José O'Connor-Smith III!!", resp.asString());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithZeroDelayUsesRandomAndReturns200() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", "0").when().get("/slowApi");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithPositiveDelayReturns200() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 1).when().get("/slowApi");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithNonIntegerDelayReturns500() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", "abc").when().get("/slowApi");
        assertEquals(500, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithNegativeDelayReturns500() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", -1).when().get("/slowApi");
        assertEquals(500, resp.getStatusCode());
    }
}
