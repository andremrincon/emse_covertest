package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;
import static org.junit.Assert.assertEquals;

public class CurrencyTest {

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = System.getProperty("api.base", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testV1AlphaAlphacode_US_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1Currency_USD_containsCurrencyCodeUSD() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/currency/USD");
        String code = resp.jsonPath().getString("[0].currencies[0].code");
        assertEquals("USD", code);
    }

    @Test(timeout = 60000)
    public void testV2Currency_EUR_fields_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/currency/EUR?fields=name;capital;population");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1Alpha_codes_query_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha?codes=US,CA,MX");
        resp.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV2AlphaAlphacode_US_fields_containsNameUnitedStates() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US?fields=name;capital;population").then().body("name", equalTo("United States of America"));
    }

    @Test(timeout = 60000)
    public void testV1Currency_numericBadRequest_400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/currency/123");
        resp.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1Alpha_alphacode_numericBadRequest_400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/123");
        resp.then().statusCode(404);
    }
}
