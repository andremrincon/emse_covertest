package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class LanguageTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testV1AlphaAlphacodeSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/{alphacode}", "US");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1AlphaAlphacodeBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/{alphacode}", "123");
        assertEquals(400, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesQuerySuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().queryParam("codes", "US,CA,MX").when().get("/v1/alpha");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesQueryNotFound() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().queryParam("codes", "true").when().get("/v1/alpha");
        assertEquals(404, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV2AlphaAlphacodeFieldsSuccess() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response act = given().queryParam("fields", "name;capital;population").when().get("/v2/alpha/{alphacode}", "US");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV2AlphaAlphacodeBadRequest() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v2/alpha/{alphacode}", "123");
        assertEquals(400, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1CurrencySuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/currency/{currency}", "USD");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1CurrencyNotFound() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/currency/{currency}", "XYZ");
        assertEquals(404, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1LangSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/lang/{lang}", "es");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1CallingCodeSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/callingcode/{callingcode}", "1");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV1NameFullTextTrueSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().queryParam("fullText", "true").when().get("/v1/name/{name}", "France");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testV2LangNotFound() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v2/lang/{lang}", "123");
        assertEquals(404, act.getStatusCode());
    }
}
