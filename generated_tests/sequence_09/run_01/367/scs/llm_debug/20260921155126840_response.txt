package ts01gpt_5_mini_3;

import org.junit.Test;
import org.junit.BeforeClass;
import static org.hamcrest.Matchers.lessThan;
import static io.restassured.RestAssured.given;
import io.restassured.response.Response;
import static org.junit.Assert.assertEquals;

public class CalcTest {
    private static String BASE;

    @BeforeClass
    public static void init() {
        String env = System.getProperty("baseUrl", System.getenv("BASE_URL") == null ? "http://localhost:8080" : System.getenv("BASE_URL"));
        BASE = env;
    }

    @Test(timeout = 60000)
    public void testAdditionReturns200() {
        given().baseUri(BASE).when().get("/api/pat/{txt}", "a").then().statusCode(lessThan(300));
        Response resp = given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "plus", "15.5", "4.5");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testSqrtProducesExpectedBody() {
        given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", "1", "algorithm").then().statusCode(lessThan(300));
        Response resp = given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "sqrt", "16", "0");
        assertEquals("4.0", resp.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testDivideByZeroProduces500() {
        given().baseUri(BASE).when().get("/api/text2txt/{word1}/{word2}/{word3}", "The", "quick", "brown").then().statusCode(lessThan(300));
        Response resp = given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "divide", "100", "0");
        assertEquals(500, resp.getStatusCode());
    }
}
