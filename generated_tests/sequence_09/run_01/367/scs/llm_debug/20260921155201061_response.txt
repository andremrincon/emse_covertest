package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class DateParseTest {

    @BeforeClass
    public static void setUp() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testDateParse_valid_Wednesday_August_returns200() {
        given().when().get("/api/calc/add/15.5/4.5").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/Wednesday/August");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateParse_valid_tuesday_MAR_returns200() {
        given().when().get("/api/cookie/session-id/abc-123-xyz-789/example.com").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/tuesday/MAR");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateParse_valid_Wednesday_MAR_returns200() {
        given().when().get("/api/text2txt/The/quick/brown").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/Wednesday/MAR");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateParse_invalid_Superday_Movember_returns500() {
        given().when().get("/api/ordered4/zebra/yak/x-ray/wolf").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/Superday/Movember");
        act.then().statusCode(500);
    }
}
