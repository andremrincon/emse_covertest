package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void test_defaultZero_when_iIsMinus4_and_sIsAbab() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/-4/abab");
        resp.then().body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void test_resultSix_when_iIsFive_and_sIsAbab() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/5/abab");
        resp.then().body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void test_status200_when_largePositiveI_triggers_upperBounds() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/700/algorithm");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void test_resultTen_when_largeNegativeI_and_algorithm_string() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/-1000/algorithm");
        resp.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_resultTen_when_iFive_and_sBaab_exercises_string_branches() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/5/baab");
        resp.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_status200_when_compareTo_equals_case_with_ababba() {
        given().when().get("/api/pat/a").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/costfuns/-4/ababba");
        resp.then().statusCode(200);
    }
}
