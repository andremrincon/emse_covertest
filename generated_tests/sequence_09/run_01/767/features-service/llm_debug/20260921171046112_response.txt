package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.junit.Assert.assertTrue;

public class ConstraintRequiresTest {

    private String getBase() {
        String env = System.getenv("BASE_URL");
        if (env != null && !env.isEmpty()) return env;
        String prop = System.getProperty("baseUrl");
        if (prop != null && !prop.isEmpty()) return prop;
        return "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProduct_returns201() {
        RestAssured.baseURI = getBase();
        String product = "Smartwatch Series 8 " + UUID.randomUUID().toString();
        Response pre = given().when().post("/products/{productName}", product);
        assertTrue(pre.getStatusCode() < 300);
        Response act = given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "Measures the oxygen saturation (SpO2) of your blood on demand.")
                .when().post("/products/{productName}/features/{featureName}", product, "Blood Oxygen Sensor");
        act.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForProduct_returns200() {
        RestAssured.baseURI = getBase();
        String product = "AeroBook-Pro-15-" + UUID.randomUUID().toString();

        assertTrue(pre1.getStatusCode() < 300);
        Response pre2 = given().contentType("application/x-www-form-urlencoded")
                .when().post("/products/{productName}/features/{featureName}", product, "16GB LPDDR5 RAM");
        assertTrue(pre2.getStatusCode() < 300);
        Response act = given().when().get("/products/{productName}/features", product);
        act.then().statusCode(200);
    }
