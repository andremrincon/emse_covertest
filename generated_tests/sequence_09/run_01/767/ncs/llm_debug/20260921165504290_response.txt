package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class TriangleClassificationTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("TEST_BASE_URL");
        if (base == null) base = System.getenv("TEST_BASE_URL");
        if (base == null) base = System.getProperty("test.api.base", "http://localhost:8080");
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void triangle_withNonPositiveSide_returns200() {
        given().when().get("/api/remainder/17/5").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/0/1/1");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void triangle_equilateral_returns200() {
        given().when().get("/api/bessj/3/2.5").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/5/5/5");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void triangle_inequalityViolation_returns200() {
        given().when().get("/api/expint/1/0.1").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/5/1/1");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void triangle_isosceles_returns200() {
        given().when().get("/api/fisher/10/5/0.75").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/5/5/3");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void triangle_scalene_returns200() {
        given().when().get("/api/gammq/5.5/2.3").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/3/4/5");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void triangle_negativeSide_returns200() {
        given().when().get("/api/bessj/0/1e-10").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/triangle/-1/2/3");
        resp.then().statusCode(200);
    }
}
