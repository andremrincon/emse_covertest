package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null) base = System.getenv("BASE_URL");
        if (base == null) base = "http://localhost:8080/rest";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testV1Alpha_US_translationsDe_assertDeValue() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha/US");
        res.then().body("translations.de", equalTo("Vereinigte Staaten von Amerika"));
    }

    @Test(timeout = 60000)
    public void testV1Alpha_invalidNumeric_returns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha/123");
        res.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV1Alpha_notFound_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha/XYZ");
        res.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1Currency_USD_translationsEs_assertEsValue() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/currency/USD");
        res.then().body("[0].translations.es", equalTo("Estados Unidos"));
    }

    @Test(timeout = 60000)
    public void testV1Currency_invalid_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/currency/XYZ");
        res.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1Name_France_translationsFr_assertFrValue() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/name/France");
        res.then().body("[0].translations.fr", equalTo("France"));
    }

    @Test(timeout = 60000)
    public void testV1Name_numeric_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/name/123");
        res.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1Callingcode_1_translationsJa_assertJaValue() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/callingcode/1");
        res.then().body("[0].translations.ja", equalTo("アメリカ合衆国"));
    }

    @Test(timeout = 60000)
    public void testV1Capital_London_status200_and_uniqueInvocation() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String unique = UUID.randomUUID().toString();
        Response res = given().when().get("/v1/capital/London");
        res.then().statusCode(200);
    }
}
