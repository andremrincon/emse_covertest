package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class LanguageTest {

    @BeforeClass
    public static void init() {
        String env = System.getenv("API_BASE");
        String prop = System.getProperty("api.base");
        String base = env != null ? env : (prop != null ? prop : "http://localhost:8080/rest");
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSetIso6391_v1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].iso639_1", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void testSetIso6392_v1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].iso639_2", equalTo("eng"));
    }

    @Test(timeout = 60000)
    public void testSetName_v1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].name", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testSetNativeName_v1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].nativeName", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testSetIso6391_v2Name_Germany() {
        given().when().get("/v2/all?fields=name;capital;region").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/name/Germany");
        resp.then().body("[0].languages[0].iso639_1", equalTo("de"));
    }

    @Test(timeout = 60000)
    public void testSetIso6392_v2Name_Germany() {
        given().when().get("/v2/all?fields=name;capital;region").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/name/Germany");
        resp.then().body("[0].languages[0].iso639_2", equalTo("deu"));
    }

    @Test(timeout = 60000)
    public void testSetName_v2Name_Germany() {
        given().when().get("/v2/all?fields=name;capital;region").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/name/Germany");
        resp.then().body("[0].languages[0].name", equalTo("German"));
    }

    @Test(timeout = 60000)
    public void testSetNativeName_v2Name_Germany() {
        given().when().get("/v2/all?fields=name;capital;region").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/name/Germany");
        resp.then().body("[0].languages[0].nativeName", equalTo("Deutsch"));
    }
}
