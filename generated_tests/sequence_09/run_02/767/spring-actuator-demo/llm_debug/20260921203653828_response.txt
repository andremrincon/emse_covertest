package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.Objects;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class SampleControllerTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSayHelloWithProvidedName() {
        String name = "María-José O'Connor-Smith III";
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("name", name).when().get("/");
        resp.then().body(equalTo("Hello " + name + "!!"));
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultNameWhenNoParamProvided() {
        given().when().get("/?name=arrange").then().statusCode(lessThan(300));
        Response resp = given().when().get("/");
        resp.then().body(equalTo("Hello Guest!!"));
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithZeroTriggersRandomDelayPath() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 0).when().get("/slowApi");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithPositiveDelaySleepsAndReturns200() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 2).when().get("/slowApi");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithInvalidDelayReturnsServerError() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", "abc").when().get("/slowApi");
        resp.then().statusCode(500);
    }
}
