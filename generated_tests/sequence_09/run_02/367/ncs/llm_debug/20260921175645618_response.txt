public class BessjTest {

    private static final String BASE = System.getProperty("baseUrl", System.getenv().getOrDefault("BASE_URL", "http://localhost:8080"));

    @Test(timeout = 60000)
    public void testBessj_axZero_returnsZero() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/3/0.0");
        resp.then().body("value", equalTo(0.0f));
    }

    @Test(timeout = 60000)
    public void testBessj_nLessThanTwo_returns400() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/1/2.5");
        resp.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testBessj_axGreaterThanN_smallAx_returns200() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/3/5.0");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessj_axGreaterThanN_largeAx_asymptotic_returns200() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/3/10.0");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessj_axLessThanN_backwardRecurrence_returns200() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/10/1e-10");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessj_negativeX_oddN_returnsNegativeValue() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(BASE + "/api/bessj/3/-5.0");
        resp.then().body("value", lessThan(0.0f));
    }
}
