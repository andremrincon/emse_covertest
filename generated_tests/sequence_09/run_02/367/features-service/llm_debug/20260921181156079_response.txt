package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CORSFilterTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testDoFilter_nonOptions_setsCorsHeadersOnGetFeatures() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/" + product + "/features");
        assertEquals("*", act.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testDoFilter_options_preflight_returnsCorsMethodHeader() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        Response act = given().when().options("/products/" + product + "/features");
        assertEquals("POST, PUT, GET, OPTIONS, DELETE", act.getHeader("Access-Control-Allow-Methods"));
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProduct_returns201() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        String feature = "feat-" + UUID.randomUUID().toString();
        Response act = given().formParam("description", "Measures the oxygen saturation (SpO2) of your blood on demand.").when().post("/products/" + product + "/features/" + feature);
        assertEquals(201, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDeleteFeature_returns204() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        String feature = "feat-" + UUID.randomUUID().toString();
        given().formParam("description", "temp").when().post("/products/" + product + "/features/" + feature).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/" + product + "/features/" + feature);
        assertEquals(204, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_returns201() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        String config = "config-" + UUID.randomUUID().toString();
        String feature = "feat-" + UUID.randomUUID().toString();
        Response act = given().when().post("/products/" + product + "/configurations/" + config + "/features/" + feature);
        assertEquals(201, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetConfigurationFeatures_setsCorsHeaderOnGet() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product).then().statusCode(lessThan(300));
        String config = "config-" + UUID.randomUUID().toString();
        given().when().post("/products/" + product + "/configurations/" + config).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/" + product + "/configurations/" + config + "/features");
        assertEquals("*", act.getHeader("Access-Control-Allow-Origin"));
    }
}
