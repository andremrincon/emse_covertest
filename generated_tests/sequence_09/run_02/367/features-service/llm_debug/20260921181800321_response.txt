package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.net.URLEncoder;
import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class ProductsConfigurationFeaturesResourceTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_success() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String product = "QuantumLeap-AI-Platform-" + uuid;
        String configuration = "standard-gpu-cluster-" + uuid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response act = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, "distributed-training");
        assertEquals(500, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_failure() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String product = "QuantumLeap-AI-Platform-" + uuid;
        String configuration = "standard-gpu-cluster-" + uuid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        String problematicFeature = "{\"feature\":\"new-feature\"}";
        String encodedFeature = URLEncoder.encode(problematicFeature, "UTF-8");
        Response act = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, encodedFeature);
        assertEquals(500, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDeleteFeature_success() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String product = "QuantumLeap-AI-Platform-" + uuid;
        String configuration = "Premium-Tier-" + uuid;
        String feature = "Realtime-Analytics-Dashboard-" + uuid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature);
        assertEquals(204, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDeleteFeature_failure() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String setupProduct = "SomeProductSetup-" + uuid;
        String setupConfiguration = "SomeConfigurationSetup-" + uuid;
        given().when().post("/products/{productName}", setupProduct).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", setupProduct, setupConfiguration).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}", "DataStreamer", "Standard", "99");
        assertEquals(500, act.getStatusCode());
    }
}
