package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.net.URLEncoder;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class RegexTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("BASE_URL");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    private static String enc(String s) {
        try {
            return URLEncoder.encode(s, "UTF-8");
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    @Test(timeout = 60000)
    public void testUrlPattern() {
        given().when().get("/api/pat/{txt}", enc("ping")).then().statusCode(lessThan(300));
        Response act = given().when().get("/api/pat/{txt}", enc("http://john/doe"));
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDatePattern() {
        given().when().get("/api/pat/{txt}", enc("healthcheck")).then().statusCode(lessThan(300));
        Response act = given().when().get("/api/pat/{txt}", enc("wed12aug"));
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testFpePattern() {
        given().when().get("/api/pat/{txt}", enc("ready")).then().statusCode(lessThan(300));
        Response act = given().when().get("/api/pat/{txt}", enc("1.2e+03"));
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testNonePattern() {
        given().when().get("/api/pat/{txt}", enc("setup")).then().statusCode(lessThan(300));
        Response act = given().when().get("/api/pat/{txt}", enc("randomtext"));
        assertEquals(200, act.getStatusCode());
    }
}
