package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class Ordered4Test {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base", System.getenv("API_BASE"));
        if (base == null) {
            base = System.getProperty("base.url", System.getenv("BASE_URL"));
        }
        if (base == null) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testIncreasing() {
        given().when().get("/api/pat/{txt}", "a").then().statusCode(lessThan(300));
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", "abcde", "bbbba", "cbcde", "dbcde")
                .then().body(equalTo("increasing"));
    }

    @Test(timeout = 60000)
    public void testDecreasing() {
        given().when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "15.5", "4.5").then().statusCode(lessThan(300));
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", "zzzzz", "yyyyy", "wwwww", "xxxxx")
                .then().body(equalTo("decreasing"));
    }

    @Test(timeout = 60000)
    public void testUnorderedDueToLengths() {
        given().when().get("/api/text2txt/{word1}/{word2}/{word3}", "The", "quick", "brown").then().statusCode(lessThan(300));
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", "four", "aaaaa", "bbbbb", "ccccc")
                .then().body(equalTo("unordered"));
    }

    @Test(timeout = 60000)
    public void testUnorderedDueToOrder() {
        given().when().get("/api/cookie/{name}/{val}/{site}", UUID.randomUUID().toString(), "1", "example.com").then().statusCode(lessThan(300));
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", "aaaaa", "aaaaa", "aaaaa", "aaaaa")
                .then().body(equalTo("unordered"));
    }

    @Test(timeout = 60000)
    public void testServerErrorStatus() {
        given().when().get("/api/pat/{txt}", "The quick brown fox jumps over the lazy dog.").then().statusCode(lessThan(300));
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", "apple", "banana", "cherry", "delta1")
                .then().statusCode(500);
    }
}
