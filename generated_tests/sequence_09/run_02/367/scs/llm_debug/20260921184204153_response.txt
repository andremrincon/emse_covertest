package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class RegexTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("BASE_URL");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testUrlPatternTriggersRegexUrlBranch() {
        String arrangeId = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", "a-" + arrangeId).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "http://john/doe");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDatePatternTriggersRegexDateBranch() {
        String arrangeId = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", arrangeId).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "wed12dec");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testFpePatternTriggersRegexFpeBranch() {
        String arrangeId = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", arrangeId).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "15.5e+02");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testNonePatternTriggersRegexNoneBranch() {
        String arrangeId = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", arrangeId).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "randomtextvalue123");
        assertEquals(200, resp.getStatusCode());
    }
}
