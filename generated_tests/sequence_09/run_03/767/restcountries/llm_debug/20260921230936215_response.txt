package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.*;
import io.restassured.RestAssured;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CountryRestV1Test {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testCORSHeaderOnV1AlphaQuery() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA,MX").when().get("/v1/alpha").then().header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void testCORSAllowedMethodsHeaderOnV1AlphaCode() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void testCORSAllowHeadersOnV1Currency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"));
    }

    @Test(timeout = 60000)
    public void testCORSCacheControlOnV1NameFullText() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fullText", "true").when().get("/v1/name/France").then().header("Cache-Control", equalTo("public, max-age=86400"));
    }

    @Test(timeout = 60000)
    public void testCORSOnV2AlphaFields() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA,MX").queryParam("fields", "name;capital;population").when().get("/v2/alpha").then().header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void testCORSOnV2AllFields() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().queryParam("fields", "name;capital;region;population;flag").when().get("/v2/all").then().header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void testCORSOnV2AlphaCodeWithFields() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().queryParam("fields", "name;capital;population").when().get("/v2/alpha/US").then().header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"));
    }

    @Test(timeout = 60000)
    public void testCORSOnV1Region() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/region/Europe").then().header("Cache-Control", equalTo("public, max-age=86400"));
    }
}
