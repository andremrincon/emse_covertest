package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CountryRestV1Test {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("BASE_URL");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080/rest";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testGetAlpha_valid_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/alpha/{alphacode}", "US");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetAlpha_invalidFormat_returns400() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/alpha/{alphacode}", "123");
        assertEquals(400, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetAlpha_notFound_returns404() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/alpha/{alphacode}", "ZZZ");
        assertEquals(404, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetAlphaList_valid_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("codes", "US,CA,MX").when().get("/v2/alpha");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetAlphaList_badRequest_invalidFormat_returns400() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("codes", "[\"US\",\"CA\"]").when().get("/v2/alpha");
        assertEquals(400, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetCurrency_valid_withFields_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("fields", "name;capital;population").when().get("/v2/currency/{currency}", "EUR");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetCurrency_badRequest_returns400() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/currency/{currency}", "123");
        assertEquals(400, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetName_fullText_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("fullText", "true").when().get("/v2/name/{name}", "Germany");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetCallingCode_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/callingcode/{callingcode}", "1");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetCapital_withFields_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("fields", "name;capital;population").when().get("/v2/capital/{capital}", "Paris");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetRegion_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/region/{region}", "Europe");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetSubRegion_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/subregion/{subregion}", "Western Europe");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetLanguage_withFields_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().queryParam("fields", "name;capital;population").when().get("/v2/lang/{lang}", "Spanish");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetDemonym_returns200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get("/v2/demonym/{demonym}", "American");
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDoPost_methodNotAllowed_returns405() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().post("/v2");
        assertEquals(405, response.getStatusCode());
    }
}
