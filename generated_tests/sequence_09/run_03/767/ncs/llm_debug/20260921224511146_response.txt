package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

public class GammqTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testGammqUsesGserWhenXLessThanAPlusOne() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/5.5/2.3");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqUsesGcfWhenXGreaterOrEqualAPlusOne() {
        given().when().get("/api/triangle/1/1/1").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/0.001/1000.0");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqReturnsResultWhenXIsZero() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/5.5/0.0");
        r.then().body("result", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGammqReturnsBadRequestForNegativeX() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/5.5/-1.0");
        r.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGammqReturnsBadRequestForNonPositiveA() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/0.0/2.0");
        r.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGammqLargeATriggersErrorPath() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/gammq/1e308/2.0");
        r.then().statusCode(400);
    }
}
