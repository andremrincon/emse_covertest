package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static org.hamcrest.Matchers.lessThan;

public class ConstraintRequiresTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProductReturns201() {
        String product = "product-" + UUID.randomUUID().toString();
        String feature = "feature-BloodOxygen-" + UUID.randomUUID().toString();
        RestAssured.given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response act = RestAssured.given().formParam("description", "Measures the oxygen saturation (SpO2) of your blood on demand.").when().post("/products/{productName}/features/{featureName}", product, feature);
        act.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationReturns201() {
        String product = "product-" + UUID.randomUUID().toString();
        String config = "config-" + UUID.randomUUID().toString();
        String feature = "feature-distributed-" + UUID.randomUUID().toString();
        RestAssured.given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        RestAssured.given().when().post("/products/{productName}/configurations/{configurationName}", product, config).then().statusCode(lessThan(300));
        Response act = RestAssured.given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature);
        act.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetConfigurationFeaturesReturns200() {
        String product = "product-" + UUID.randomUUID().toString();
        String config = "config-" + UUID.randomUUID().toString();
        String feature = "feature-" + UUID.randomUUID().toString();
        RestAssured.given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        RestAssured.given().when().post("/products/{productName}/configurations/{configurationName}", product, config).then().statusCode(lessThan(300));
        RestAssured.given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature).then().statusCode(lessThan(300));
        Response act = RestAssured.given().when().get("/products/{productName}/configurations/{configurationName}/features", product, config);
        act.then().statusCode(200);
    }
}
