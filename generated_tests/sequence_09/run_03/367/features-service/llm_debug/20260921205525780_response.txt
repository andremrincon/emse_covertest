package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.hasSize;
import static org.hamcrest.Matchers.lessThan;

public class ProductsConfigurationsServiceTest {

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = System.getenv().getOrDefault("API_BASE_URL", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testGetConfigurationsNamesForProduct_singleConfiguration() {
        String uid = UUID.randomUUID().toString();
        String product = "Laptop-Pro-" + uid;
        String configuration = "cfg-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/{productName}/configurations", product);
        act.then().body("$", hasItem(configuration));
    }

    @Test(timeout = 60000)
    public void testGetConfigurationActivedFeaturesNames_afterAddFeature() {
        String uid = UUID.randomUUID().toString();
        String product = "QuantumLeap-" + uid;
        String configuration = "standard-" + uid;
        String feature = "distributed-training-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/{productName}/configurations/{configurationName}/features", product, configuration);
        act.then().body("$", hasItem(feature));
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_success_status201() {
        String uid = UUID.randomUUID().toString();
        String product = "Platform-" + uid;
        String configuration = "conf-" + uid;
        String feature = "feature-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response act = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature);
        act.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_duplicate_throws500() {
        String uid = UUID.randomUUID().toString();
        String product = "DupProd-" + uid;
        String configuration = "DupConf-" + uid;
        String feature = "dup-feature-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature).then().statusCode(lessThan(300));
        Response act = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature);
        act.then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testRemoveFeatureFromConfiguration_success_status204() {
        String uid = UUID.randomUUID().toString();
        String product = "RemProd-" + uid;
        String configuration = "RemConf-" + uid;
        String feature = "rem-feature-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, feature);
        act.then().statusCode(204);
    }

    @Test(timeout = 60000)
    public void testEvaluateAndUpdateConfiguration_invalid_setsValidFalse() {
        String uid = UUID.randomUUID().toString();
        String product = "EvalProdInvalid-" + uid;
        String configuration = "EvalConf-" + uid;
        String sourceFeature = "A-" + uid;
        String requiredFeature = "B-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded").formParam("sourceFeature", sourceFeature).formParam("requiredFeature", requiredFeature).when().post("/products/{productName}/constraints/requires", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, sourceFeature).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/{productName}/configurations/{configurationName}", product, configuration);
        act.then().body("valid", equalTo(false));
    }

    @Test(timeout = 60000)
    public void testEvaluateAndUpdateConfiguration_valid_setsValidTrue() {
        String uid = UUID.randomUUID().toString();
        String product = "EvalProdValid-" + uid;
        String configuration = "EvalConfV-" + uid;
        String sourceFeature = "A2-" + uid;
        String requiredFeature = "B2-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded").formParam("sourceFeature", sourceFeature).formParam("requiredFeature", requiredFeature).when().post("/products/{productName}/constraints/requires", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, requiredFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, configuration, sourceFeature).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/{productName}/configurations/{configurationName}", product, configuration);
        act.then().body("valid", equalTo(true));
    }

    @Test(timeout = 60000)
    public void testGetConfigurationActivedFeaturesNames_empty_when_none() {
        String uid = UUID.randomUUID().toString();
        String product = "EmptyFeatProd-" + uid;
        String configuration = "EmptyFeatConf-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, configuration).then().statusCode(lessThan(300));
        Response act = given().when().get("/products/{productName}/configurations/{configurationName}/features", product, configuration);
        act.then().body("$", hasSize(0));
    }
}
