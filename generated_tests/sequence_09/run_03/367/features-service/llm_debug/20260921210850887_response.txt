package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ProductsConfigurationFeaturesResourceTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) base = System.getenv("API_BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationSuccess() {
        String uuid = UUID.randomUUID().toString();
        String productName = "QuantumLeap-AI-Platform-" + uuid;
        String configurationName = "standard-gpu-cluster-" + uuid;
        String featureName = "distributed-training";
        given().pathParam("productName", productName).when().post("/products/{productName}").then().statusCode(lessThan(300));
        given().pathParams("productName", productName, "configurationName", configurationName).when().post("/products/{productName}/configurations/{configurationName}").then().statusCode(lessThan(300));
        Response act = given().pathParams("productName", productName, "configurationName", configurationName, "featureName", featureName).when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}");
        act.then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationFailure() {
        String uuid = UUID.randomUUID().toString();
        String productName = "QuantumLeap-AI-Platform-" + uuid;
        String configurationName = "standard-gpu-cluster-" + uuid;
        String featureName = "feature-name-that-is-intentionally-made-extremely-long-to-exceed-any-reasonable-database-column-width-or-url-path-segment-limit-and-potentially-cause-an-unhandled-exception-or-a-buffer-overflow-somewhere-deep-in-the-application-stack-resulting-in-a-generic-five-hundred-internal-server-error-response-instead-of-a-more-graceful-four-hundred-bad-request-error";
        given().pathParam("productName", productName).when().post("/products/{productName}").then().statusCode(lessThan(300));
        given().pathParams("productName", productName, "configurationName", configurationName).when().post("/products/{productName}/configurations/{configurationName}").then().statusCode(lessThan(300));
        Response act = given().pathParams("productName", productName, "configurationName", configurationName, "featureName", featureName).when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}");
        act.then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testDeleteFeatureSuccess() {
        String uuid = UUID.randomUUID().toString();
        String productName = "QuantumLeap-AI-" + uuid;
        String configurationName = "Premium-Tier-" + uuid;
        String featureName = "Realtime-Analytics-Dashboard";
        given().pathParam("productName", productName).when().post("/products/{productName}").then().statusCode(lessThan(300));
        given().pathParams("productName", productName, "configurationName", configurationName).when().post("/products/{productName}/configurations/{configurationName}").then().statusCode(lessThan(300));
        Response act = given().pathParams("productName", productName, "configurationName", configurationName, "featureName", featureName).when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}");
        act.then().statusCode(204);
    }

    @Test(timeout = 60000)
    public void testDeleteFeatureFailure() {
        String uuid = UUID.randomUUID().toString();
        String productName = "DataStreamer-" + uuid;
        String configurationName = "Standard-" + uuid;
        String featureName = "99";
        given().pathParam("productName", productName).when().post("/products/{productName}").then().statusCode(lessThan(300));
        given().pathParams("productName", productName, "configurationName", configurationName).when().post("/products/{productName}/configurations/{configurationName}").then().statusCode(lessThan(300));
        Response act = given().pathParams("productName", productName, "configurationName", configurationName, "featureName", featureName).when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}");
        act.then().statusCode(500);
    }
}
