package ts01gpt_5_mini_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static org.hamcrest.Matchers.lessThan;

public class ConstraintRequiresTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base", System.getenv().getOrDefault("API_BASE", "http://localhost"));
        String port = System.getProperty("api.port", System.getenv().getOrDefault("API_PORT", "8080"));
        RestAssured.baseURI = base;
        try {
            RestAssured.port = Integer.parseInt(port);
        } catch (NumberFormatException e) {
            RestAssured.port = 8080;
        }
        RestAssured.basePath = System.getProperty("api.basePath", "/");
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProduct_returns201() {
        String product = "prod-" + UUID.randomUUID();
        String feature = "feat-" + UUID.randomUUID();
        RestAssured.given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response act = RestAssured.given().formParam("description", "Measures the oxygen saturation (SpO2) of your blood on demand.").when().post("/products/{productName}/features/{featureName}", product, feature);
        act.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForProduct_returns200() {
        String product = "prod-" + UUID.randomUUID();
        String feature = "feat-" + UUID.randomUUID();
        RestAssured.given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        RestAssured.given().formParam("description", "desc").when().post("/products/{productName}/features/{featureName}", product, feature).then().statusCode(lessThan(300));
        Response act = RestAssured.given().when().get("/products/{productName}/features", product);
        act.then().statusCode(200);
    }

        RestAssured.given().formParam("description", "to be deleted").when().post("/products/{productName}/features/{featureName}", product, feature).then().statusCode(lessThan(300));
        Response act = RestAssured.given().when().delete("/products/{productName}/features/{featureName}", product, feature);
        act.then().statusCode(204);
    }

        Response act = RestAssured.given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature);
        act.then().statusCode(201);
