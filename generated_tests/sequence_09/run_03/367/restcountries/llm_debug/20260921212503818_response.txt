package ts01gpt_5_mini_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base", System.getenv().getOrDefault("API_BASE", "http://localhost:8080/rest"));
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testGetAlphaByCode_Status200_US() {
        given().when().get("/v1/all").then().statusCode(404);
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetAlphaByCode_TranslationsDe_US() {
        given().when().get("/v1/all").then().statusCode(404);
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("translations.de", equalTo("Vereinigte Staaten von Amerika"));
    }

    @Test(timeout = 60000)
    public void testGetAlphaByCode_TranslationsEs_US() {
        given().when().get("/v1/alpha?codes=US,CA,MX").then().statusCode(404);
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("translations.es", equalTo("Estados Unidos"));
    }

    @Test(timeout = 60000)
    public void testGetNameByName_TranslationsFr_France() {
        given().when().get("/v1/alpha?codes=DE,FR").then().statusCode(404);
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].translations.fr", equalTo("France"));
    }

    @Test(timeout = 60000)
    public void testGetNameByName_TranslationsIt_France() {
        given().when().get("/v1/all").then().statusCode(404);
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].translations.it", equalTo("Francia"));
    }

    @Test(timeout = 60000)
    public void testGetAlphaByCode_TranslationsJa_US() {
        given().when().get("/v1/alpha?codes=US,CA").then().statusCode(404);
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("translations.ja", equalTo("アメリカ合衆国"));
    }
}
