package ts01gpt_5_mini_2;

import org.junit.Test;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    @Test(timeout = 60000)
    public void testCostfuns_exampleAlgorithm_status200() {
        String base = System.getProperty("API_BASE_URL", System.getenv("API_BASE_URL"));
        if (base == null) base = "http://localhost:8080";
        given().baseUri(base).when().get("/api/calc/add/15.5/4.5").then().statusCode(lessThan(300));
        Response act = given().baseUri(base).when().get("/api/costfuns/{i}/{s}", 1, "algorithm");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfuns_negativeLarge_status200() {
        String base = System.getProperty("API_BASE_URL", System.getenv("API_BASE_URL"));
        if (base == null) base = "http://localhost:8080";
        given().baseUri(base).when().get("/api/calc/subtract/15.5/4.5").then().statusCode(lessThan(300));
        Response act = given().baseUri(base).when().get("/api/costfuns/{i}/{s}", -500, "abab");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfuns_largeValues_compareZeroCase_status200() {
        String base = System.getProperty("API_BASE_URL", System.getenv("API_BASE_URL"));
        if (base == null) base = "http://localhost:8080";
        given().baseUri(base).when().get("/api/calc/add/0/4.5").then().statusCode(lessThan(300));
        Response act = given().baseUri(base).when().get("/api/costfuns/{i}/{s}", 700, "ababba");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfuns_iMinus4_bodyZero() {
        String base = System.getProperty("API_BASE_URL", System.getenv("API_BASE_URL"));
        if (base == null) base = "http://localhost:8080";
        given().baseUri(base).when().get("/api/pat/{txt}", "The quick brown fox jumps over the lazy dog.").then().statusCode(lessThan(300));
        Response act = given().baseUri(base).when().get("/api/costfuns/{i}/{s}", -4, "abab");
        act.then().body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_sBaab_status200() {
        String base = System.getProperty("API_BASE_URL", System.getenv("API_BASE_URL"));
        if (base == null) base = "http://localhost:8080";
        given().baseUri(base).when().get("/api/text2txt/{word1}/{word2}/{word3}", "The", "quick", "brown").then().statusCode(lessThan(300));
        Response act = given().baseUri(base).when().get("/api/costfuns/{i}/{s}", -4, "baab");
        act.then().statusCode(200);
    }
}
