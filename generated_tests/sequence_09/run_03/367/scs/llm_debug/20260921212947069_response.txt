package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class DateParseTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.baseurl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASEURL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testDateparse_mon_jan_returns200() {
        given().when().get("/api/pat/{txt}", "The").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "feb").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "mar").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/{dayname}/{monthname}", "mon", "jan");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateparse_tue_feb_returns200() {
        given().when().get("/api/pat/{txt}", "a").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "apr").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "may").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/{dayname}/{monthname}", "tue", "feb");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateparse_wed_mar_returns200() {
        given().when().get("/api/title/{sex}/{title}", "male", "Smith").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "jun").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "jul").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "mar");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateparse_thur_dec_returns200() {
        given().when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "15.5", "4.5").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "aug").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{dayname}/{monthname}", "wed", "sep").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/{dayname}/{monthname}", "thur", "dec");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateparse_invalid_Monday_Movember_returns500() {
        given().when().get("/api/pat/{txt}", "The quick brown fox jumps over the lazy dog.").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/dateparse/{dayname}/{monthname}", "Monday", "Movember");
        act.then().statusCode(500);
    }
}
