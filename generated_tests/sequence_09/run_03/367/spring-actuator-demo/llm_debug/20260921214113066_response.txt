package ts01gpt_5_mini_2;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import java.util.UUID;
import static org.junit.Assert.assertEquals;

public class SampleControllerTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSayHelloWithName() {
        String uniqueName = "John Smith " + UUID.randomUUID().toString();
        given().queryParam("name", uniqueName).when().get("/").then().statusCode(lessThan(300));
        Response act = given().queryParam("name", uniqueName).when().get("/");
        assertEquals("Hello " + uniqueName + "!!", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultGuest() {
        String marker = "A-" + UUID.randomUUID().toString();
        given().queryParam("name", marker).when().get("/").then().statusCode(lessThan(300));
        Response act = given().when().get("/");
        assertEquals("Hello Guest!!", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithExplicitDelay() {
        given().when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", 1).when().get("/slowApi").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithZeroTriggersRandomDelay() {
        given().when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", 0).when().get("/slowApi").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithNegativeDelayReturnsServerError() {
        given().when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", -1).when().get("/slowApi").then().statusCode(500);
    }
}
