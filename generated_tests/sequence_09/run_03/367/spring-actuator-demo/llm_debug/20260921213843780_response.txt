package ts01gpt_5_mini;

import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class SampleControllerTest {

    private static final String BASE_URL = System.getProperty("baseUrl", System.getenv().getOrDefault("BASE_URL", "http://localhost:8080"));

    @Test(timeout = 60000)
    public void testSayHelloWithProvidedName() {
        String name = "María-José O'Connor-Smith III";
        given().baseUri(BASE_URL).when().get("/?name=" + UUID.randomUUID().toString()).then().statusCode(lessThan(300));
        given().baseUri(BASE_URL).when().get("/?name=" + java.net.URLEncoder.encode(name)).then().body(equalTo("Hello " + name + "!!"));
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultGuest() {
        given().baseUri(BASE_URL).when().get("/?name=" + UUID.randomUUID().toString()).then().statusCode(lessThan(300));
        given().baseUri(BASE_URL).when().get("/").then().body(equalTo("Hello Guest!!"));
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithZeroDelayTriggersRandomPath() {
        given().baseUri(BASE_URL).when().get("/?name=" + UUID.randomUUID().toString()).then().statusCode(lessThan(300));
        given().baseUri(BASE_URL).when().get("/slowApi?delay=0").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithPositiveDelayReturnsResultBody() {
        given().baseUri(BASE_URL).when().get("/?name=" + UUID.randomUUID().toString()).then().statusCode(lessThan(300));
        given().baseUri(BASE_URL).when().get("/slowApi?delay=1").then().body(equalTo("Result"));
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithNegativeDelayReturnsServerError() {
        given().baseUri(BASE_URL).when().get("/?name=" + UUID.randomUUID().toString()).then().statusCode(lessThan(300));
        given().baseUri(BASE_URL).when().get("/slowApi?delay=-1").then().statusCode(500);
    }

}
