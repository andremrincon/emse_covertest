package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class CountryRestV2Test {

    @BeforeClass
    public static void setUp() {
        String base = System.getProperty("base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_Success_US_returns200() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_BadRequest_shortCode_returns400() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/A").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_NotFound_zzz_returns404() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/ZZZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_WithFields_returnsFilteredBody() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US?fields=name;capital;population").then().body(containsString("name"));
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_Success_codesUSCA_returns200() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha?codes=US,CA").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_BadRequest_arrayFormat_returns400() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha?codes=%5B%22US%22,%20%22CA%22%5D").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_Success_EUR_returns200() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/currency/EUR").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_BadRequest_invalidLength_returns400() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/currency/12").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByName_Success_Germany_fullTextFalse_returns200() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/name/Germany?fullText=false").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByName_NotFound_numeric_returns404() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/name/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByCallingCode_Success_1_returns200() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/callingcode/1").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCallingCode_NotFound_999999_returns404() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/callingcode/999999").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetAllV2_WithFields_returnsFilteredCountriesBody() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/all?fields=name;capital;region;population;flag").then().body(containsString("name"));
    }

    @Test(timeout = 60000)
    public void testDoPOST_V2_methodNotAllowed_returns405() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().post("/v2").then().statusCode(405);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_MissingCodes_paramNull_returns400() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha").then().statusCode(400);
    }
}
