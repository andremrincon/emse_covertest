package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @BeforeClass
    public static void setup() {
        String url = System.getProperty("API_BASE_URL");
        if (url == null) {
            url = System.getenv("API_BASE_URL");
        }
        if (url == null || url.isEmpty()) {
            url = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = url;
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowOriginOnRootGet() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/");
        resp.then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowMethodsOnV1AlphaQuery() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().queryParam("codes", "US,CA,MX").when().get("/v1/alpha");
        resp.then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowHeadersOnV1AlphaCode() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnV1Currency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/currency/USD");
        resp.then().header("Cache-Control", "public, max-age=86400");
    }

    @Test(timeout = 60000)
    public void testCORSOnV2AllWithFields() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        Response resp = given().queryParam("fields", "name;capital;region;population;flag").when().get("/v2/all");
        resp.then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testCORSOnContributePost() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().contentType("application/json").body("{\"amount\":1,\"currency\":\"USD\",\"token\":\"tok_123\"}").when().post("/contribute");
        resp.then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowMethodsOnV2AlphaWithFields() {
        given().when().get("/v2").then().statusCode(lessThan(300));
        Response resp = given().queryParam("fields", "name;capital;population").when().get("/v2/alpha/US");
        resp.then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testCORSOnV1NameWithFullTextParam() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().queryParam("fullText", "false").when().get("/v1/name/France");
        resp.then().header("Access-Control-Allow-Origin", "*");
    }
}
