package ts01gpt_5_mini_3;

import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class RegexTest {

    private static final String BASE;
    static {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            String env = System.getenv("API_BASE");
            base = (env == null || env.isEmpty()) ? "http://localhost:8080" : env;
        }
        BASE = base;
    }

    @Test(timeout = 60000)
    public void testPat_PatLengthTwo_ReturnsZero() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "health-" + uuid).then().statusCode(lessThan(300));
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", "irrelevant-" + uuid, "ab").then().body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPat_PatFound_NoReverse_ReturnsOne() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "ready-" + uuid).then().statusCode(lessThan(300));
        String txt = "xxabcyy-" + uuid;
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", txt, "abc").then().body(equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testPat_ReverseFoundOnly_ReturnsTwo() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "prep-" + uuid).then().statusCode(lessThan(300));
        String txt = "zzcbayy-" + uuid;
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", txt, "abc").then().body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testPat_PatFollowedImmediatelyByReverse_ReturnsIndex() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "ping-" + uuid).then().statusCode(lessThan(300));
        String pat = "abc";
        String patrev = "cba";
        String txt = "xx" + pat + patrev + "yy-" + uuid;
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", txt, pat).then().body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testPat_ReverseFollowedImmediatelyByPat_ReturnsIndex() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "check-" + uuid).then().statusCode(lessThan(300));
        String pat = "abc";
        String patrev = "cba";
        String txt = "zz" + patrev + pat + "aa-" + uuid;
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", txt, pat).then().body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testPat_PatAndReverseNonAdjacent_ReturnsFirstIndex() {
        String uuid = UUID.randomUUID().toString();
        given().baseUri(BASE).when().get("/api/pat/{txt}", "start-" + uuid).then().statusCode(lessThan(300));
        String pat = "abc";
        String patrev = "cba";
        String txt = pat + "XX" + patrev + "-" + uuid;
        given().baseUri(BASE).when().get("/api/pat/{txt}/{pat}", txt, pat).then().body(equalTo("0"));
    }
}
