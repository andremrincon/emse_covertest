package ts01glm_5_2;

import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import java.util.UUID;
import org.junit.BeforeClass;
import org.junit.Test;

public class NotFoundExceptionMapperTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testNotFoundExceptionMapperReturns404ForNonExistentPath() {
        given()
        .when()
            .get("/v1/all")
        .then()
            .statusCode(lessThan(300));

        given()
        .when()
            .get("/v3/" + UUID.randomUUID().toString())
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testNotFoundExceptionMapperReturns404ForExtraPathSegment() {
        given()
        .when()
            .get("/v2/all")
        .then()
            .statusCode(lessThan(300));

        given()
        .when()
            .get("/v1/alpha/US/" + UUID.randomUUID().toString())
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testNotFoundExceptionMapperReturns404ForUnknownResourcePath() {
        given()
        .when()
            .get("/v1/all")
        .then()
            .statusCode(lessThan(300));

        given()
        .when()
            .get("/nonexistent-" + UUID.randomUUID().toString())
        .then()
            .statusCode(404);
    }
}
