package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCORSFilterAddsAllowOriginHeaderOnV1All() {
        given()
                .when()
                .get("/v1/all")
                .then()
                .statusCode(200)
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testCORSFilterAddsAllowMethodsHeaderOnV1Alpha() {
        given()
                .queryParam("codes", "US")
                .when()
                .get("/v1/alpha")
                .then()
                .header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testCORSFilterAddsAllowHeadersHeaderOnV1AlphaCode() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()

                .header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
