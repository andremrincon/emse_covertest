package ts01glm_5_2_3;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class Ordered4Test {

    @BeforeClass
    public static void setUp() {
        String host = System.getProperty("server.host", "localhost");
        String port = System.getProperty("server.port", "8080");
        RestAssured.baseURI = "http://" + host + ":" + port;
    }

    @Test(timeout = 60000)
    public void testOrdered4Increasing() {
        String w = "aaaaa";
        String x = "bbbbb";
        String z = "ddddd";
        String y = "ccccc";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("increasing"));
    }

    @Test(timeout = 60000)
    public void testOrdered4Decreasing() {
        String w = "ddddd";
        String x = "ccccc";
        String z = "aaaaa";
        String y = "bbbbb";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("decreasing"));
    }

    @Test(timeout = 60000)
    public void testOrdered4UnorderedValidLengthsNoOrder() {
        String w = "aaaaa";
        String x = "aaaaa";
        String z = "aaaaa";
        String y = "aaaaa";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("unordered"));
    }

    @Test(timeout = 60000)
    public void testOrdered4UnorderedShortLength() {
        String w = "a";
        String x = "bbbbb";
        String z = "ccccc";
        String y = "ddddd";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("unordered"));
    }

    @Test(timeout = 60000)
    public void testOrdered4UnorderedLongLength() {
        String w = "aaaaaaa";
        String x = "bbbbb";
        String z = "ccccc";
        String y = "ddddd";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("unordered"));
    }

    @Test(timeout = 60000)
    public void testOrdered4UnorderedPartialIncreasing() {
        String w = "aaaaa";
        String x = "bbbbb";
        String z = "eeeee";
        String y = "ddddd";

        given().when()
                .get("/api/ordered4/" + w + "/" + x + "/" + z + "/" + y)
                .then()
                .statusCode(200)
                .body(equalTo("unordered"));
    }
}
