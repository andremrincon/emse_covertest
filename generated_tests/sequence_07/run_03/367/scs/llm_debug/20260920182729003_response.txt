package ts01glm_5_2_3;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

public class CalcTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
        RestAssured.enableLoggingOfRequestAndResponseIfValidationFails();
    }

    @Test(timeout = 60000)
    public void testPatFoundReverseImmediatelyAfterPalindrome() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/ABABBABA/ABAB")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatFoundReverseFoundLaterInText() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/ABABXBABA/ABAB")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testReverseFoundPatImmediatelyAfterPalindrome() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/BABAABAB/ABAB")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testReverseFoundPatFoundLaterInText() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/BABAXABAB/ABAB")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatFoundButReverseNotFound() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/ABABXYZW/ABAB")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNeitherPatNorReverseFoundInText() {
        RestAssured.given()
            .accept(ContentType.TEXT)
        .when()
            .get("/api/pat/XYZXYZXY/ABAB")
        .then()
            .statusCode(200);
    }
}
