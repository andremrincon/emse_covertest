package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class Ordered4Test {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testIncreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/alpha/bravo/gamma/delta")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDecreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/zebra/tiger/mouse/snake")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedAllSame() {
        given()
            .when()
                .get("/api/ordered4/alpha/alpha/alpha/alpha")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedWTooShort() {
        given()
            .when()
                .get("/api/ordered4/abcd/bravo/gamma/delta")
