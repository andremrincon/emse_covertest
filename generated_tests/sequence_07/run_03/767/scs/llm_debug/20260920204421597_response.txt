package ts01glm_5_2_2;

import io.restassured.RestAssured;
import static io.restassured.RestAssured.*;
import org.junit.Before;
import org.junit.Test;

public class Text2TxtTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testForWordAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/for/the/way")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFourWordAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/four/the/way")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSeeYouAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/see/you/soon")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testByTheWayAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/by/the/way")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSeeWithoutYouFallsThrough() {
        given()
            .when()
            .get("/api/text2txt/see/hello/soon")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testByTheWithoutWayFallsThrough() {
        given()
            .when()
            .get("/api/text2txt/by/the/wrong")
            .then()
            .statusCode(200);
    }
}
