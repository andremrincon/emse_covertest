package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class GammqTest {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGcfPathLargeX() {
        given()
            .when()
                .get("/api/gammq/1.0/1000.0")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserNormalConvergence() {
        given()
            .when()
                .get("/api/gammq/5.5/2.3")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserXZeroBoundary() {
        given()
            .when()
                .get("/api/gammq/5.5/0.0")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGcfSmallA() {
        given()
            .when()
                .get("/api/gammq/0.001/2.3")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testInvalidNegativeA() {
        given()
            .when()
                .get("/api/gammq/-1.0/2.3")
            .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testInvalidAType() {
        given()
            .when()
                .get("/api/gammq/abc/2.3")
            .then()
                .statusCode(400);
    }
}
