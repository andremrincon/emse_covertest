package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("baseUrl");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
        RestAssured.enableLoggingOfRequestAndResponseIfValidationFails();
    }

    @Test(timeout = 60000)
    public void testAlphaCodeUsReturnsTranslations() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(404)
                .body("translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testAlphaCodeGbReturnsEsTranslation() {
        given()
            .when()
                .get("/v1/alpha/GB")
            .then()
                .statusCode(404)
                .body("translations.es", notNullValue());
    }

    @Test(timeout = 60000)
    public void testNameFranceReturnsFrTranslation() {
        given()
            .when()
                .get("/v1/name/France")
            .then()
                .statusCode(404)
                .body("[0].translations.fr", notNullValue());
    }

    @Test(timeout = 60000)
    public void testNameUsaReturnsJaTranslation() {

    @Test(timeout = 60000)
    public void testAlphaCodesQueryParamReturnsItTranslation() {
        given()
            .queryParam("codes", "US,CA,MX")
            .when()
                .get("/v1/alpha")
            .then()
                .statusCode(404)
                .body("[0].translations.it", notNullValue());
    }

    @Test(timeout = 60000)
    public void testAllV1ReturnsTranslationsDe() {

            .when()
                .get("/v1/all")
            .then()
                .statusCode(404)
                .body("[0].translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testAlphaCodeUsWithFullTranslationsObject() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(404)
                .body("translations", hasKey("de"))
                .body("translations", hasKey("es"))
                .body("translations", hasKey("fr"))
                .body("translations", hasKey("ja"))
                .body("translations", hasKey("it"));
    }

    public void testCurrencyXofReturnsTranslationsFr() {
