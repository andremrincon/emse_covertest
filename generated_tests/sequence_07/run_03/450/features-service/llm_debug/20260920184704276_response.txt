    @Test(timeout = 60000)
    public void testDeleteRequiresConstraint() {
        String productName = "tp-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "sf-" + UUID.randomUUID().toString().substring(0, 8);
        String requiredFeature = "rf-" + UUID.randomUUID().toString().substring(0, 8);

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeature).then().statusCode(lessThan(300));
        given()
            .contentType(ContentType.URLENC)
            .formParam("sourceFeature", sourceFeature)
            .formParam("requiredFeature", requiredFeature)
        .when().post("/products/{productName}/constraints/requires", productName).then().statusCode(lessThan(300));

        Response productResponse = given().when().get("/products/{productName}", productName);
        productResponse.then().statusCode(lessThan(300));
        List<Map<String, Object>> constraints = productResponse.jsonPath().getList("constraints");
        Integer constraintId = null;
        if (constraints != null) {
            for (Map<String, Object> constraint : constraints) {
                if ("requires".equals(constraint.get("type"))) {
                    Object id = constraint.get("id");
                    if (id instanceof Integer) {
                        constraintId = (Integer) id;
                    }
                    break;
                }
            }
        }

        if (constraintId != null) {
            given()
                .when()
                .delete("/products/{productName}/constraints/{constraintId}", productName, constraintId)
                .then()
                .statusCode(204);
        }
    }
