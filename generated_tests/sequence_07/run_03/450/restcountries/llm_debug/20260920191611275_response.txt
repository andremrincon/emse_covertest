package ts01glm_5_2_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertNull;

public class CORSFilterTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null) {
            baseUrl = System.getenv("baseUrl");
        }
        if (baseUrl == null) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowOriginHeaderOnV1All() {
        Response resp = given()
            .when()
                .get("/v1/all")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowMethodsHeaderOnV1Alpha() {
        Response resp = given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Methods"));
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowHeadersHeaderOnV1Name() {
        Response resp = given()
            .when()
                .get("/v1/name/France")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Headers"));
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnV1Region() {
        Response resp = given()
            .when()
                .get("/v1/region/Europe")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Cache-Control"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnBadRequestResponse() {
        Response resp = given()
            .when()
                .get("/v1/alpha?codes=123")
            .then()
                .statusCode(400)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnNotFoundResponse() {
        Response resp = given()
            .when()
                .get("/v1/alpha/XYZ")
            .then()
                .statusCode(404)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnServerErrorResponse() {
        Response resp = given()
            .when()
                .get("/v1/name/True")
            .then()
                .statusCode(500)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnV2All() {
        Response resp = given()
            .when()
                .get("/v2/all")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testResponseChainPassesRequestBody() {
        given()
            .when()
                .get("/v1/callingcode/1")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnCapitalEndpoint() {
        Response resp = given()
            .when()
                .get("/v1/capital/London")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnCurrencyEndpoint() {
        Response resp = given()
            .when()
                .get("/v1/currency/USD")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnSubregionEndpoint() {
        Response resp = given()
            .when()
                .get("/v1/subregion/Western%20Europe")
            .then()
                .statusCode(200)
                .extract().response();
        assertNull(resp.getHeader("Access-Control-Allow-Origin"));
    }
}
