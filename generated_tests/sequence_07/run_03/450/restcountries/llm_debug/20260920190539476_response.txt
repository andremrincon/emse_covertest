                .when()

    @Test(timeout = 60000)
    public void testV1AllReturnsLanguageFields() {
        JsonPath response = given()
                .when()
                .get("/v1/all")
                .then()
                .statusCode(200)
                .extract()
                .jsonPath();

        List<Map<String, Object>> countries = response.getList("");
        assertNotNull(countries);
        assertFalse(countries.isEmpty());

        boolean foundLanguage = false;
        for (Map<String, Object> country : countries) {
            List<Map<String, Object>> languages = (List<Map<String, Object>>) country.get("languages");
            if (languages != null && !languages.isEmpty()) {
                Map<String, Object> lang = languages.get(0);
                if (lang.get("iso639_1") != null && lang.get("iso639_2") != null
                        && lang.get("name") != null && lang.get("nativeName") != null) {
                    foundLanguage = true;
                    break;
                }
            }
        }
        assertTrue("At least one country should have fully populated language fields", foundLanguage);
    }
