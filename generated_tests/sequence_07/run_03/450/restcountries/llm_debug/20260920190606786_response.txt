package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class NotFoundExceptionMapperTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl != null && !baseUrl.isEmpty()) {
            RestAssured.baseURI = baseUrl;
        } else {
            RestAssured.baseURI = "http://localhost:8080/rest";
        }
    }

    @Test(timeout = 60000)
    public void testNotFoundMapperForUnknownV1Path() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/v1/nonexistentpathabc")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testNotFoundMapperForNonExistentRootPath() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/nonexistentrootxyz")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testNotFoundMapperForUnknownV2Path() {
        given()
            .when()
                .get("/v2/all")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/v2/nonexistentpathdef")
            .then()
                .statusCode(404);
    }
}
