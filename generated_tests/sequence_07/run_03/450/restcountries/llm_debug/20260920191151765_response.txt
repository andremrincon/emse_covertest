package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.notNullValue;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV1AllReturnsDeTranslation() {
        given().when().get("/v1/all").then().body("[0].translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsEsTranslation() {
        given().when().get("/v1/alpha/{alphacode}", "US").then().body("translations.es", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsFrTranslation() {
        given().when().get("/v1/alpha/{alphacode}", "GB").then().body("translations.fr", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesReturnsJaTranslation() {
        given().queryParam("codes", "US,CA,MX").when().get("/v1/alpha").then().body("[0].translations.ja", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1CurrencyReturnsItTranslation() {
        given().when().get("/v1/currency/{currency}", "USD").then().body("[0].translations.it", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1NameReturnsDeTranslation() {
        given().when().get("/v1/name/{name}", "France").then().body("[0].translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1CallingCodeReturnsEsTranslation() {
        given().when().get("/v1/callingcode/{callingcode}", "1").then().body("[0].translations.es", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1CapitalReturnsFrTranslation() {
        given().when().get("/v1/capital/{capital}", "London").then().body("[0].translations.fr", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1RegionReturnsJaTranslation() {
        given().when().get("/v1/region/{region}", "Europe").then().body("[0].translations.ja", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1SubregionReturnsItTranslation() {
        given().when().get("/v1/subregion/{subregion}", "Western Europe").then().body("[0].translations.it", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1LangReturnsDeTranslation() {
        given().when().get("/v1/lang/{lang}", "es").then().body("[0].translations.de", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaInvalidCodeReturns400() {
        given().when().get("/v1/alpha/{alphacode}", "123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV1AlphaNotFoundReturns404() {
        given().when().get("/v1/alpha/{alphacode}", "XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1CurrencyNotFoundReturns404() {
        given().when().get("/v1/currency/{currency}", "XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1NameNotFoundReturns404() {
        given().when().get("/v1/name/{name}", "123").then().statusCode(404);
    }
}
