package ts01glm_5_2_3;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.response.Response;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;

public class LanguageTest {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void v1AlphaUsReturnsLanguageIso639_1() {
        Response r = given()
        .when()
            .get(baseUrl + "/v1/alpha/US")
        .then()
            .statusCode(lessThan(300))
            .extract().response();
        String iso = r.path("languages[0].iso639_1");
        assertEquals("en", iso);
    }

    @Test(timeout = 60000)
    public void v1AlphaGbReturnsLanguageIso639_2() {

        .then()
            .statusCode(lessThan(300))
