package ts01glm_5_2_3;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;

public class CountryRestV1Test {

    @BeforeClass
    public static void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl",
                System.getenv().getOrDefault("BASE_URL", "http://localhost:8080/rest"));
    }

    @Test(timeout = 60000)
    public void contributeWithNullBodyReturnsBadRequest() {
        given()
            .contentType(ContentType.JSON)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void contributeWithBlankTokenReturnsBadRequest() {
        given()
            .contentType(ContentType.JSON)
            .body("{\"amount\":100,\"token\":\"  \"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void contributeWithValidTokenEntersChargeFlowAndReturnsBadRequest() {
        String uniqueToken = "tok_" + UUID.randomUUID().toString().replace("-", "");
        given()
            .contentType(ContentType.JSON)
            .body("{\"amount\":500,\"token\":\"" + uniqueToken + "\"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }
}
