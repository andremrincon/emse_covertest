package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.containsString;
import static org.hamcrest.Matchers.lessThan;

public class Ordered4Test {

    @BeforeClass
    public static void setUp() {
        String host = System.getProperty("host", "localhost");
        String port = System.getProperty("port", "8080");
        RestAssured.baseURI = "http://" + host + ":" + port;
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenLengthCheckFails() {
        given()
            .when()
                .get("/api/ordered4/a/b/d/c")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/a/b/d/c")
            .then()
                .statusCode(200)
                .body(containsString("unordered"));
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenOneStringTooLong() {
        given()
            .when()
                .get("/api/ordered4/aaaaaaa/bbbbb/ddddd/ccccc")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/aaaaaaa/bbbbb/ddddd/ccccc")
            .then()
                .statusCode(200)
                .body(containsString("unordered"));
    }

    @Test(timeout = 60000)
    public void testIncreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/ddddd/ccccc")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/ddddd/ccccc")
            .then()
                .statusCode(200)
                .body(containsString("increasing"));
    }

    @Test(timeout = 60000)
    public void testDecreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/ddddd/ccccc/aaaaa/bbbbb")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/ddddd/ccccc/aaaaa/bbbbb")
            .then()
                .statusCode(200)
                .body(containsString("decreasing"));
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenLengthPassesButNotOrdered() {
        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/ccccc/bbbbb")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/ccccc/bbbbb")
            .then()
                .statusCode(200)
                .body(containsString("unordered"));
    }

    @Test(timeout = 60000)
    public void testIncreasingWithBoundaryLengths5And6() {
        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/dddddd/ccccc")
            .then()
                .statusCode(lessThan(300));

        given()
            .when()
                .get("/api/ordered4/aaaaa/bbbbb/dddddd/ccccc")
            .then()
                .statusCode(200)
                .body(containsString("increasing"));
    }
}
