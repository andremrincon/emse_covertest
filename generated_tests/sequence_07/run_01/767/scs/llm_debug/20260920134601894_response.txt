package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class Ordered4Test {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testUnorderedLengthCheckFailsShortStrings() {
        given()
            .pathParam("w", "a")
            .pathParam("x", "b")
            .pathParam("z", "c")
            .pathParam("y", "d")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testIncreasingOrderAllLengthFive() {
        given()
            .pathParam("w", "aaaaa")
            .pathParam("x", "bbbbb")
            .pathParam("z", "ddddd")
            .pathParam("y", "ccccc")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDecreasingOrderAllLengthFive() {
        given()
            .pathParam("w", "ddddd")
            .pathParam("x", "ccccc")
            .pathParam("z", "aaaaa")
            .pathParam("y", "bbbbb")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedLengthPassesNeitherIncreasingNorDecreasing() {
        given()
            .pathParam("w", "bbbbb")
            .pathParam("x", "aaaaa")
            .pathParam("z", "ddddd")
            .pathParam("y", "ccccc")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testUnorderedLengthCheckFailsLongStrings() {
        given()
            .pathParam("w", "aaaaaaa")
            .pathParam("x", "bbbbbbb")
            .pathParam("z", "ccccccc")
            .pathParam("y", "ddddddd")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDecreasingOrderBoundaryLengthFiveAndSix() {
        given()
            .pathParam("w", "zzzzzz")
            .pathParam("x", "yyyyy")
            .pathParam("z", "wwwww")
            .pathParam("y", "xxxxxx")
        .when()
            .get("/api/ordered4/{w}/{x}/{z}/{y}")
        .then()
            .statusCode(200);
    }
}
