    private static int sendRequest(String method, String path, String contentType, byte[] body) throws IOException {
        String full = path.startsWith("/") ? baseUrl + path : baseUrl + "/" + path;
        URL url = new URL(full);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(60000);
        conn.setReadTimeout(60000);
        if (contentType != null) {
            conn.setRequestProperty("Content-Type", contentType);
        }
        if (body != null && (method.equals("POST") || method.equals("PUT") || method.equals("PATCH"))) {
            conn.setDoOutput(true);
            conn.setRequestProperty("Content-Length", String.valueOf(body.length));
            OutputStream os = null;
            try {
                os = conn.getOutputStream();
                os.write(body);
                os.flush();
            } finally {
                if (os != null) {
                    try { os.close(); } catch (IOException ignored) {}
                }
            }
        }
        int code;
        InputStream is = null;
        try {
            code = conn.getResponseCode();
            is = conn.getInputStream();
            byte[] buffer = new byte[1024];
            while (is.read(buffer) != -1) {
            }
        } catch (IOException e) {
            try {
                code = conn.getResponseCode();
                InputStream es = conn.getErrorStream();
                if (es != null) {
                    byte[] buffer = new byte[1024];
                    try {
                        while (es.read(buffer) != -1) {
                        }
                    } finally {
                        try { es.close(); } catch (IOException ignored) {}
                    }
                }
            } catch (IOException e2) {
                throw e;
            }
        } finally {
            if (is != null) {
                try { is.close(); } catch (IOException ignored) {}
            }
            conn.disconnect();
        }
        return code;
    }

    private static byte[] encodeForm(Map<String, String> params) throws IOException {
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : params.entrySet()) {
            if (!first) {
                sb.append("&");
            }
            first = false;
            sb.append(URLEncoder.encode(e.getKey(), "UTF-8"));
            sb.append("=");
            sb.append(URLEncoder.encode(e.getValue(), "UTF-8"));
        }
        return sb.toString().getBytes("UTF-8");
    }

    private static int post(String path) throws IOException {
        return sendRequest("POST", path, null, null);
    }

    private static int postForm(String path, Map<String, String> form) throws IOException {
        byte[] body = encodeForm(form);
        return sendRequest("POST", path, "application/x-www-form-urlencoded", body);
    }

    private static int putForm(String path, Map<String, String> form) throws IOException {
        byte[] body = encodeForm(form);
        return sendRequest("PUT", path, "application/x-www-form-urlencoded", body);
    }

    private static int delete(String path) throws IOException {
        return sendRequest("DELETE", path, null, null);
    }

    private static int get(String path) throws IOException {
        return sendRequest("GET", path, null, null);
    }
