    private String baseURI;

    @Before
    public void setUp() {
        String env = System.getenv("BASE_URL");
        if (env != null && !env.isEmpty()) {
            baseURI = env;
        } else {
            baseURI = "http://localhost:8080";
        }
    }

    private String formatPath(String template, Object... values) {
        String s = template;
        for (Object v : values) {
            String encoded;
            try {
                encoded = URLEncoder.encode(String.valueOf(v), "UTF-8");
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
            s = s.replaceFirst("\\{[^}]+\\}", encoded);
        }
        return s;
    }

    private int httpPost(String pathTemplate, String contentType, String body, Object... pathParams) throws IOException {
        String fullPath = baseURI + formatPath(pathTemplate, pathParams);
        URL url = new URL(fullPath);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        try {
            conn.setRequestMethod("POST");
            conn.setConnectTimeout(15000);
            conn.setReadTimeout(15000);
            if (contentType != null) {
                conn.setRequestProperty("Content-Type", contentType);
            }
            conn.setDoOutput(true);
            if (body != null) {
                byte[] bytes = body.getBytes("UTF-8");
                conn.setRequestProperty("Content-Length", String.valueOf(bytes.length));
                OutputStream os = conn.getOutputStream();
                try {
                    os.write(bytes);
                    os.flush();
                } finally {
                    os.close();
                }
            } else {
                conn.setRequestProperty("Content-Length", "0");
                try {
                    OutputStream os = conn.getOutputStream();
                    os.close();
                } catch (IOException ignore) {
                }
            }
            int code = conn.getResponseCode();
            drainStream(conn);
            return code;
        } finally {
            conn.disconnect();
        }
    }

    private int httpGet(String pathTemplate, Object... pathParams) throws IOException {
        String fullPath = baseURI + formatPath(pathTemplate, pathParams);
        URL url = new URL(fullPath);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        try {
            conn.setRequestMethod("GET");
            conn.setConnectTimeout(15000);
            conn.setReadTimeout(15000);
            int code = conn.getResponseCode();
            drainStream(conn);
            return code;
        } finally {
            conn.disconnect();
        }
    }

    private String buildForm(Map<String, String> form) {
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : form.entrySet()) {
            if (!first) sb.append("&");
            first = false;
            try {
                sb.append(URLEncoder.encode(e.getKey(), "UTF-8"));
                sb.append("=");
                sb.append(URLEncoder.encode(e.getValue(), "UTF-8"));
            } catch (Exception ex) {
                throw new RuntimeException(ex);
            }
        }
        return sb.toString();
    }

    private void drainStream(HttpURLConnection conn) {
        InputStream is = null;
        try {
            try {
                is = conn.getInputStream();
            } catch (IOException e) {
                is = conn.getErrorStream();
            }
            if (is != null) {
                Scanner s = new Scanner(is, "UTF-8");
                while (s.hasNextLine()) {
                    s.nextLine();
                }
                s.close();
            }
        } finally {
            if (is != null) {
                try {
                    is.close();
                } catch (IOException ignore) {
                }
            }
        }
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationWithNoConstraints() throws Exception {
        String productName = "eval-no-constraints-" + UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "base-config";

        int code;
        code = httpPost("/products/{productName}", null, null, productName);
        Assert.assertTrue(code < 300);
        code = httpPost("/products/{productName}/configurations/{configurationName}", null, null, productName, configurationName);
        Assert.assertTrue(code < 300);

        code = httpGet("/products/{productName}/configurations/{configurationName}/features", productName, configurationName);
        Assert.assertEquals(200, code);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationWithRequiresConstraintNoDerivedFeatures() throws Exception {
        String productName = "eval-req-noderived-" + UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "base-config";
        String sourceFeature = "FeatureA";
        String requiredFeature = "FeatureB";

        int code;
        code = httpPost("/products/{productName}", null, null, productName);
        Assert.assertTrue(code < 300);
        code = httpPost("/products/{productName}/features/{featureName}", null, null, productName, sourceFeature);
        Assert.assertTrue(code < 300);
        code = httpPost("/products/{productName}/features/{featureName}", null, null, productName, requiredFeature);
        Assert.assertTrue(code < 300);
        code = httpPost("/products/{productName}/configurations/{configurationName}", null, null, productName, configurationName);
        Assert.assertTrue(code < 300);

        Map<String, String> form = new HashMap<String, String>();
        form.put("sourceFeature", sourceFeature);
        form.put("requiredFeature", requiredFeature);
        String body = buildForm(form);
        code = httpPost("/products/{productName}/constraints/requires", "application/x-www-form-urlencoded", body, productName);
        Assert.assertTrue(code < 300);

        code = httpGet("/products/{productName}/configurations/{configurationName}/features", productName, configurationName);
        Assert.assertEquals(200, code);
    }
