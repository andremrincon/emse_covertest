    private int send(String method, String pathTemplate, Map<String, String> formParams, Object... pathParams) {
        try {
            String fullUrl = buildUrl(pathTemplate, pathParams);
            URL url = new URL(fullUrl);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod(method);
            if (formParams != null && (method.equals("POST") || method.equals("PUT"))) {
                conn.setDoOutput(true);
                String body = buildFormBody(formParams);
                byte[] bytes = body.getBytes("UTF-8");
                conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded; charset=UTF-8");
                conn.setRequestProperty("Content-Length", Integer.toString(bytes.length));
                OutputStream os = conn.getOutputStream();
                os.write(bytes);
                os.flush();
                os.close();
            } else {
                conn.setDoOutput(false);
            }
            int status = conn.getResponseCode();
            conn.disconnect();
            return status;
        } catch (Exception e) {
            fail("Request failed: " + e.getMessage());
            return -1;
        }
    }

    private String buildUrl(String template, Object... params) throws Exception {
        String path = template;
        if (params != null) {
            for (Object p : params) {
                String replacement = URLEncoder.encode(String.valueOf(p), "UTF-8");
                path = path.replaceFirst("\\{[^/}]*\\}", replacement);
            }
        }
        if (!path.startsWith("/")) {
            path = "/" + path;
        }
        return baseUrl + path;
    }

    private String buildFormBody(Map<String, String> formParams) throws Exception {
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : formParams.entrySet()) {
            if (!first) {
                sb.append("&");
            }
            first = false;
            sb.append(URLEncoder.encode(e.getKey(), "UTF-8"));
            sb.append("=");
            sb.append(URLEncoder.encode(e.getValue(), "UTF-8"));
        }
        return sb.toString();
    }
