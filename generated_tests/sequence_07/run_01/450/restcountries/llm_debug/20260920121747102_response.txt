package ts01glm_5_2_3;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

public class CurrencyTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("baseUrl");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV1AllReturnsCurrencies() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/all").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsCurrency() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().statusCode(200).body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaMultipleCodesReturnsCurrencies() {
        given().when().queryParam("codes", "US,CA,MX").get("/v1/alpha").then().statusCode(lessThan(300));
        given().when().queryParam("codes", "US,CA,MX").get("/v1/alpha").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1CurrencyEndpointReturnsCountriesWithCurrencies() {
        given().when().get("/v1/currency/USD").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1NameReturnsCurrencies() {
        given().when().get("/v1/name/France").then().statusCode(lessThan(300));
        given().when().get("/v1/name/France").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1RegionReturnsCurrencies() {
        given().when().get("/v1/region/Europe").then().statusCode(lessThan(300));
        given().when().get("/v1/region/Europe").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCurrencies() {
        given().when().get("/v2/currency/EUR").then().statusCode(lessThan(300));
        given().when().get("/v2/currency/EUR").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2RegionalblocReturnsCurrencies() {
        given().when().get("/v2/regionalbloc/EU").then().statusCode(lessThan(300));
        given().when().get("/v2/regionalbloc/EU").then().statusCode(200).body("[0].currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrency() {
        given().when().get("/v2/alpha/US").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US").then().statusCode(200).body("currencies", notNullValue());
    }
}
