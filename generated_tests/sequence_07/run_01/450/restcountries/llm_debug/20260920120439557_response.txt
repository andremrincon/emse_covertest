package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryServiceTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseURI", "http://localhost:8080");
        RestAssured.basePath = System.getProperty("basePath", "/rest");
    }

    @Test(timeout = 60000)
    public void getByLanguage_twoCharCode_returnsCountries() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/lang/es")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByLanguage_threeCharCode_returnsCountries() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/lang/spa")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByLanguage_invalidThreeCharCode_returnsNotFound() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/lang/123")
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByRegionalBloc_acronymEU_returnsCountries() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/regionalbloc/EU")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByRegionalBloc_acronymNAFTA_returnsCountries() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/regionalbloc/NAFTA")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByRegionalBloc_invalidAcronym_returnsNotFound() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/regionalbloc/123")
        .then()
            .statusCode(404);
    }
}
