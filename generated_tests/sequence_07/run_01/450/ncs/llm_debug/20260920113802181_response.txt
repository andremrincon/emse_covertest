package ts01glm_5_2_3;

import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class RemainderTest {

    @Before
    public void setUp() {
        baseURI = System.getProperty("baseURI", "http://localhost");
        port = Integer.parseInt(System.getProperty("port", "8080"));
        basePath = System.getProperty("basePath", "");
    }

    @Test(timeout = 60000)
    public void testRemainderAZero() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/0/5")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderBZero() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/17/0")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderBothPositive() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/17/5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveA_NegativeB() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/17/-5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeA_PositiveB() {
        given()
            .accept("application/json")

            .statusCode(200);

    @Test(timeout = 60000)
    public void testRemainderBothNegative() {

}
