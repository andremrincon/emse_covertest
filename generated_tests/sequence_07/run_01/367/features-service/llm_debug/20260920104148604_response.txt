    private static class Response {
        final int statusCode;
        final String body;
        final Map<String, List<String>> headers;

        Response(int statusCode, String body, Map<String, List<String>> headers) {
            this.statusCode = statusCode;
            this.body = body;
            this.headers = headers;
        }

        String header(String name) {
            if (name == null) return null;
            List<String> vals = headers.get(name);
            if (vals == null) {
                for (Map.Entry<String, List<String>> e : headers.entrySet()) {
                    if (name.equalsIgnoreCase(e.getKey())) {
                        vals = e.getValue();
                        break;
                    }
                }
            }
            if (vals == null || vals.isEmpty()) return null;
            return vals.get(0);
        }
    }

    private static String encode(String s) {
        try {
            return URLEncoder.encode(s, "UTF-8");
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private static String formatPath(String template, Object... args) {
        String result = template;
        for (Object arg : args) {
            String encoded = encode(String.valueOf(arg));
            result = result.replaceFirst("\\{[^}]+\\}", encoded);
        }
        return result;
    }

    private static String formEncode(Map<String, String> params) {
        if (params == null || params.isEmpty()) return "";
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : params.entrySet()) {
            if (!first) sb.append("&");
            first = false;
            sb.append(encode(e.getKey())).append("=").append(encode(e.getValue()));
        }
        return sb.toString();
    }

    private static Response request(String method, String pathTemplate, Map<String, String> formParams, String contentType, Object... args) throws IOException {
        String path = formatPath(pathTemplate, args);
        URL url = new URL(baseUri + path);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(30000);
        conn.setReadTimeout(30000);
        if (contentType != null) {
            conn.setRequestProperty("Content-Type", contentType);
        }
        if (formParams != null && !formParams.isEmpty()) {
            String body = formEncode(formParams);
            conn.setDoOutput(true);
            byte[] bytes = body.getBytes(Charset.forName("UTF-8"));
            conn.setRequestProperty("Content-Length", Integer.toString(bytes.length));
            DataOutputStream dos = new DataOutputStream(conn.getOutputStream());
            try {
                dos.write(bytes);
                dos.flush();
            } finally {
                dos.close();
            }
        }
        int status = conn.getResponseCode();
        InputStream is;
        try {
            if (status >= 200 && status < 400) {
                is = conn.getInputStream();
            } else {
                is = conn.getErrorStream();
                if (is == null) {
                    is = conn.getInputStream();
                }
            }
        } catch (IOException e) {
            is = conn.getErrorStream();
            if (is == null) {
                throw e;
            }
        }
        String body = "";
        if (is != null) {
            BufferedReader br = new BufferedReader(new InputStreamReader(is, Charset.forName("UTF-8")));
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) {
                sb.append(line);
            }
            body = sb.toString();
            try {
                br.close();
            } catch (IOException ignored) {}
        }
        Map<String, List<String>> headers = conn.getHeaderFields();
        conn.disconnect();
        return new Response(status, body, headers == null ? new HashMap<String, List<String>>() : headers);
    }

    private static Response post(String pathTemplate, Object... args) throws IOException {
        return request("POST", pathTemplate, null, null, args);
    }

    private static Response postForm(String pathTemplate, Map<String, String> formParams, Object... args) throws IOException {
        return request("POST", pathTemplate, formParams, "application/x-www-form-urlencoded", args);
    }

    private static Response putForm(String pathTemplate, Map<String, String> formParams, Object... args) throws IOException {
        return request("PUT", pathTemplate, formParams, "application/x-www-form-urlencoded", args);
    }

    private static Response delete(String pathTemplate, Object... args) throws IOException {
        return request("DELETE", pathTemplate, null, null, args);
    }

    private static Response get(String pathTemplate, Object... args) throws IOException {
        return request("GET", pathTemplate, null, null, args);
    }

    private static int countOccurrences(String haystack, String needle) {
        if (haystack == null || haystack.isEmpty() || needle == null || needle.isEmpty()) return 0;
        int count = 0;
        int idx = 0;
        while ((idx = haystack.indexOf(needle, idx)) != -1) {
            count++;
            idx += needle.length();
        }
        return count;
    }
