    private static class Response {
        final int status;
        final String body;

        Response(int status, String body) {
            this.status = status;
            this.body = body;
        }
    }

    private String resolvePath(String path, Object... params) {
        String result = path;
        for (Object param : params) {
            String encoded;
            try {
                encoded = URLEncoder.encode(String.valueOf(param), "UTF-8");
            } catch (UnsupportedEncodingException e) {
                encoded = String.valueOf(param);
            }
            result = result.replaceFirst("\\{[^/}]+\\}", encoded);
        }
        return result;
    }

    private Response doRequest(String method, String path, Object... params) {
        HttpURLConnection conn = null;
        try {
            String resolved = resolvePath(path, params);
            URL url = new URL(baseUrl + resolved);
            conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod(method);
            conn.setConnectTimeout(10000);
            conn.setReadTimeout(10000);
            conn.setDoInput(true);
            int status = conn.getResponseCode();
            InputStream is = null;
            try {
                if (status >= 200 && status < 400) {
                    is = conn.getInputStream();
                } else {
                    is = conn.getErrorStream();
                }
            } catch (IOException e) {
                is = conn.getErrorStream();
            }
            String body = "";
            if (is != null) {
                BufferedReader br = new BufferedReader(new InputStreamReader(is, "UTF-8"));
                StringBuilder sb = new StringBuilder();
                String line;
                while ((line = br.readLine()) != null) {
                    sb.append(line);
                }
                body = sb.toString();
                br.close();
            }
            return new Response(status, body);
        } catch (IOException e) {
            throw new RuntimeException(e);
        } finally {
            if (conn != null) {
                conn.disconnect();
            }
        }
    }

    private void assertStatusLessThan(String method, String path, int upperExclusive, Object... params) {
        Response r = doRequest(method, path, params);
        assertTrue("Expected status < " + upperExclusive + " but was " + r.status, r.status < upperExclusive);
    }

    private void assertStatusEquals(String method, String path, int expected, Object... params) {
        Response r = doRequest(method, path, params);
        assertTrue("Expected status == " + expected + " but was " + r.status, r.status == expected);
    }

    private void assertBodyContains(String method, String path, Object[] params, String item, boolean shouldContain) {
        Response r = doRequest(method, path, params);
        if (shouldContain) {
            assertTrue("Response body did not contain expected item: " + item + " Body: " + r.body, r.body != null && r.body.contains(item));
        } else {
            assertFalse("Response body contained item but should not: " + item + " Body: " + r.body, r.body != null && r.body.contains(item));
        }
    }
