package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ContributionTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void contributeWithValidAmountAndTokenReturnsAccepted() {
        String payload = "{\"amount\":1000,\"currency\":\"USD\",\"token\":\"tok_visa\"}";

        given()
                .contentType(ContentType.JSON)
                .body(payload)
        .when()
                .post("/contribute")
        .then()
                .statusCode(202);
    }

    @Test(timeout = 60000)
    public void contributeWithMissingAmountReturnsBadRequest() {
        String payload = "{\"currency\":\"USD\",\"token\":\"tok_visa\"}";

                .contentType(ContentType.JSON)
                .body(payload)
        .when()
                .post("/contribute")
        .then()
                .statusCode(400);
    }
