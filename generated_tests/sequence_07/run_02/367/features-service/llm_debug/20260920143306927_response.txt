package ts01glm_5_2_3;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.is;

public class ProductsConfigurationsServiceTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    private String uniqueProductName() {
        return "TestProduct-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String uniqueConfigName() {
        return "TestConfig-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String uniqueFeatureName() {
        return "TestFeature-" + UUID.randomUUID().toString().substring(0, 8);
    }

    @Test(timeout = 60000)
    public void getConfigurationsNamesForProductReturnsConfigurationNames() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));

        given().when().get("/products/{productName}/configurations", productName)
                .then().statusCode(200).body("$", hasItem(configName));
    }

    @Test(timeout = 60000)
    public void getConfigurationsNamesForProductReturnsEmptyListWhenNoConfigurations() {
        String productName = uniqueProductName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));

        given().when().get("/products/{productName}/configurations", productName)
                .then().statusCode(200).body("size()", is(0));
    }

    @Test(timeout = 60000)
    public void getConfigurationActivedFeaturesNamesReturnsActiveFeatures() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String featureName = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, featureName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName).then().statusCode(lessThan(300));

        given().when().get("/products/{productName}/configurations/{configurationName}/features", productName, configName)
                .then().statusCode(200).body("$", hasItem(featureName));
    }

    @Test(timeout = 60000)

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));

        given().when().get("/products/{productName}/configurations/{configurationName}/features", productName, configName)
                .then().statusCode(200).body("size()", is(0));
    }

    @Test(timeout = 60000)
    public void addFeatureFromConfigurationSucceedsAndEvaluatesConfiguration() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String featureName = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, featureName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));

        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName)
                .then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void addFeatureFromConfigurationThrowsDuplicatedWhenFeatureAlreadyActive() {
        String productName = uniqueProductName();

        String featureName = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, featureName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName).then().statusCode(lessThan(300));

        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName)
                .then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void removeFeatureFromConfigurationSucceedsAndEvaluatesConfiguration() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String featureName = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, featureName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName).then().statusCode(lessThan(300));

        given().when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName)
                .then().statusCode(204);
    }

    @Test(timeout = 60000)

    @Test(timeout = 60000)
    public void evaluateAndUpdateConfigurationSetsValidWhenNoConstraints() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String featureName = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, featureName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, featureName).then().statusCode(lessThan(300));

        given().when().get("/products/{productName}/configurations/{configurationName}", productName, configName)

    @Test(timeout = 60000)
    public void evaluateAndUpdateConfigurationSetsInvalidWhenConstraintViolated() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String sourceFeature = uniqueFeatureName();
        String excludedFeature = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, excludedFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().contentType(ContentType.URLENC)
                .formParam("sourceFeature", sourceFeature)
                .formParam("excludedFeature", excludedFeature)
                .when().post("/products/{productName}/constraints/excludes", productName)
                .then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, excludedFeature).then().statusCode(500);

        given().when().get("/products/{productName}/configurations/{configurationName}", productName, configName)
                .then().statusCode(200).body("valid", is(false));
    }

    @Test(timeout = 60000)
    public void addFeatureFromConfigurationEvaluatesWithRequiresConstraintValid() {
        String productName = uniqueProductName();
        String configName = uniqueConfigName();
        String sourceFeature = uniqueFeatureName();
        String requiredFeature = uniqueFeatureName();

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().contentType(ContentType.URLENC)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when().post("/products/{productName}/constraints/requires", productName)
                .then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, sourceFeature).then().statusCode(lessThan(300));
