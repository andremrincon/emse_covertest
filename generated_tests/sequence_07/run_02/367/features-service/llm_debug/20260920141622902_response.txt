package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import io.restassured.path.json.JsonPath;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class ProductsConstraintsResourceTest {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    private String uniqueProduct() {
        return "TestProduct-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private void createProduct(String productName) {
        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
    }

    private void addFeature(String productName, String featureName) {
        given().contentType(ContentType.URLENC).when()
            .post("/products/{productName}/features/{featureName}", productName, featureName)
            .then().statusCode(lessThan(300));
    }

    private void addConfiguration(String productName, String configName) {
        given().when()
            .post("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(lessThan(300));
    }

    private void addFeatureToConfiguration(String productName, String configName, String featureName) {
        given().when()
            .post("/products/{productName}/configurations/{configurationName}/features/{featureName}",
                productName, configName, featureName)
            .then().statusCode(lessThan(300));
    }

    private void addExcludesConstraint(String productName, String sourceFeature, String excludedFeature) {
        given().contentType(ContentType.URLENC)
            .formParam("sourceFeature", sourceFeature)
            .formParam("excludedFeature", excludedFeature)
            .when()
            .post("/products/{productName}/constraints/excludes", productName)
            .then().statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testCreateExcludesConstraint() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "FeatureA");
        addFeature(productName, "FeatureB");

        given().contentType(ContentType.URLENC)
            .formParam("sourceFeature", "FeatureA")
            .formParam("excludedFeature", "FeatureB")
            .when()
            .post("/products/{productName}/constraints/excludes", productName)
            .then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationBothFeaturesActive() {
        String productName = uniqueProduct();
        String configName = "config-" + UUID.randomUUID().toString().substring(0, 8);
        createProduct(productName);
        addFeature(productName, "SourceFeat");
        addFeature(productName, "ExcludedFeat");
        addExcludesConstraint(productName, "SourceFeat", "ExcludedFeat");
        addConfiguration(productName, configName);
        addFeatureToConfiguration(productName, configName, "SourceFeat");
        addFeatureToConfiguration(productName, configName, "ExcludedFeat");

        given().when()
            .get("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(200).body("valid", is(false));
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationOnlySourceActive() {
        String productName = uniqueProduct();
        String configName = "config-" + UUID.randomUUID().toString().substring(0, 8);
        createProduct(productName);
        addFeature(productName, "SourceFeat");
        addFeature(productName, "ExcludedFeat");
        addExcludesConstraint(productName, "SourceFeat", "ExcludedFeat");
        addConfiguration(productName, configName);
        addFeatureToConfiguration(productName, configName, "SourceFeat");

        given().when()
            .get("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(200).body("valid", is(true));
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationOnlyExcludedActive() {

        addFeature(productName, "SourceFeat");
        addFeature(productName, "ExcludedFeat");
        addExcludesConstraint(productName, "SourceFeat", "ExcludedFeat");

            .get("/products/{productName}/configurations/{configurationName}", productName, configName)

    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationNeitherFeatureActive() {
        String productName = uniqueProduct();
        String configName = "config-" + UUID.randomUUID().toString().substring(0, 8);
        createProduct(productName);
        addFeature(productName, "SourceFeat");
        addFeature(productName, "ExcludedFeat");
        addFeature(productName, "OtherFeat");
        addExcludesConstraint(productName, "SourceFeat", "ExcludedFeat");
        addConfiguration(productName, configName);
        addFeatureToConfiguration(productName, configName, "OtherFeat");

        given().when()
            .get("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(200).body("valid", is(true));
    }

    @Test(timeout = 60000)
    public void testGetTypeReturnsExcludes() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "FeatX");
        addFeature(productName, "FeatY");
        addExcludesConstraint(productName, "FeatX", "FeatY");

        given().when()
            .get("/products/{productName}", productName)
            .then().statusCode(200).body("constraints", hasItem(hasEntry("type", "excludes")));
    }

    @Test(timeout = 60000)
    public void testGetSourceFeatureNameViaProduct() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "SourceFeatXYZ");
        addFeature(productName, "ExcludedFeatXYZ");
        addExcludesConstraint(productName, "SourceFeatXYZ", "ExcludedFeatXYZ");

        given().when()
            .get("/products/{productName}", productName)
            .then().statusCode(200).body("constraints", hasItem(hasEntry("sourceFeatureName", "SourceFeatXYZ")));
    }

    public void testGetExcludedFeatureNameViaProduct() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "SourceFeatABC");
        addFeature(productName, "ExcludedFeatABC");
        addExcludesConstraint(productName, "SourceFeatABC", "ExcludedFeatABC");

    }

    @Test(timeout = 60000)
    public void testDeleteExcludesConstraint() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "FeatDel1");
        addFeature(productName, "FeatDel2");
        addExcludesConstraint(productName, "FeatDel1", "FeatDel2");

        String body = given().when().get("/products/{productName}", productName).asString();
        Integer constraintId = JsonPath.from(body).getInt("constraints.find { it.type == 'excludes' }.id");

        given().when()
            .delete("/products/{productName}/constraints/{constraintId}", productName, constraintId)
            .then().statusCode(204);
    }

    public void testSetSourceFeatureNameWithDifferentValue() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "SrcFeat1");

    @Test(timeout = 60000)
    public void testEvaluateConfigurationWithMultipleExcludesConstraints() {
        String productName = uniqueProduct();
        String configName = "config-" + UUID.randomUUID().toString().substring(0, 8);
        createProduct(productName);
        addFeature(productName, "FeatA");
        addFeature(productName, "FeatB");
        addFeature(productName, "FeatC");
        addExcludesConstraint(productName, "FeatA", "FeatB");
        addExcludesConstraint(productName, "FeatB", "FeatC");
        addConfiguration(productName, configName);
        addFeatureToConfiguration(productName, configName, "FeatA");
        addFeatureToConfiguration(productName, configName, "FeatC");

        given().when()
            .get("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(200).body("valid", is(true));
    }

    @Test(timeout = 60000)
    public void testDefaultConstructorViaExcludesConstraintCreation() {
        String productName = uniqueProduct();
        createProduct(productName);
        addFeature(productName, "FeatInit1");
        addFeature(productName, "FeatInit2");

        given().contentType(ContentType.URLENC)
            .formParam("sourceFeature", "FeatInit1")
            .formParam("excludedFeature", "FeatInit2")
            .when()
            .post("/products/{productName}/constraints/excludes", productName)
            .then().statusCode(201).body("constraintType", equalTo("EXCLUDES"));
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationBothActiveReturnsErrorMessage() {
        String productName = uniqueProduct();
        String configName = "config-" + UUID.randomUUID().toString().substring(0, 8);
        createProduct(productName);
        addFeature(productName, "SrcErr");
        addFeature(productName, "ExcErr");
        addExcludesConstraint(productName, "SrcErr", "ExcErr");
        addConfiguration(productName, configName);
        addFeatureToConfiguration(productName, configName, "SrcErr");
        addFeatureToConfiguration(productName, configName, "ExcErr");

        given().when()
            .get("/products/{productName}/configurations/{configurationName}", productName, configName)
            .then().statusCode(200).body("valid", is(false));
    }
}
