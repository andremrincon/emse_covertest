import java.io.OutputStream;

public class ProductsConstraintsResourceTest {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        if (baseUrl.endsWith("/")) {
            baseUrl = baseUrl.substring(0, baseUrl.length() - 1);
        }
    }

    private String uniqueProductName() {
        return "test-prod-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String uniqueFeatureName() {
        return "test-feat-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String uniqueConfigName() {
        return "test-cfg-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String buildPath(String template, Object... args) {
        String result = template;
        for (Object arg : args) {
            int start = result.indexOf('{');
            int end = result.indexOf('}', start);
            if (start == -1 || end == -1) {
                break;
            }
            String encoded = urlEncode(String.valueOf(arg));
            result = result.substring(0, start) + encoded + result.substring(end + 1);
        }
        return result;
    }

    private String urlEncode(String s) {
        try {
            return URLEncoder.encode(s, "UTF-8").replace("+", "%20");
        } catch (UnsupportedEncodingException e) {
            return s;
        }
    }

    private int sendRequest(String method, String pathTemplate, Map<String, String> formParams, Object... pathArgs) throws IOException {
        String path = buildPath(pathTemplate, pathArgs);
        String fullUrl = baseUrl + path;
        URL url = new URL(fullUrl);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(30000);
        conn.setReadTimeout(30000);
        if ("POST".equalsIgnoreCase(method) && formParams != null && !formParams.isEmpty()) {
            conn.setDoOutput(true);
            conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded; charset=UTF-8");
            String body = formEncode(formParams);
            byte[] bytes = body.getBytes("UTF-8");
            conn.setRequestProperty("Content-Length", String.valueOf(bytes.length));
            OutputStream os = null;
            try {
                os = conn.getOutputStream();
                os.write(bytes);
                os.flush();
            } finally {
                if (os != null) {
                    try { os.close(); } catch (IOException ignored) {}
                }
            }
        }
        int status;
        try {
            status = conn.getResponseCode();
            InputStream is = (status >= 200 && status < 400) ? conn.getInputStream() : conn.getErrorStream();
            if (is != null) {
                try {
                    Scanner s = new Scanner(is, "UTF-8");
                    while (s.hasNextLine()) {
                        s.nextLine();
                    }
                    s.close();
                } finally {
                    try { is.close(); } catch (IOException ignored) {}
                }
            }
        } finally {
            conn.disconnect();
        }
        return status;
    }

    private String formEncode(Map<String, String> params) {
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : params.entrySet()) {
            if (!first) {
                sb.append("&");
            }
            first = false;
            sb.append(urlEncode(e.getKey())).append("=").append(urlEncode(e.getValue()));
        }
        return sb.toString();
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProductSuccess() throws Exception {
        String productName = uniqueProductName();
        String featureName = uniqueFeatureName();

        int status1 = sendRequest("POST", "/products/{productName}", null, productName);
        Assert.assertTrue(status1 < 300);

        Map<String, String> params = new LinkedHashMap<String, String>();
        params.put("description", "Test feature description");
        int status2 = sendRequest("POST", "/products/{productName}/features/{featureName}", params, productName, featureName);
        Assert.assertEquals(201, status2);
    }

        String featureName = uniqueFeatureName();

        int s1 = sendRequest("POST", "/products/{productName}", null, productName);
        Assert.assertTrue(s1 < 300);

        int s2 = sendRequest("POST", "/products/{productName}/features/{featureName}", p1, productName, featureName);
        Assert.assertTrue(s2 < 300);

        Map<String, String> p2 = new LinkedHashMap<String, String>();
        p2.put("description", "Second description");
        int s3 = sendRequest("POST", "/products/{productName}/features/{featureName}", p2, productName, featureName);
        Assert.assertEquals(500, s3);
    }

        Assert.assertTrue(s1 < 300);

        p1.put("description", "Test feature");
        int s2 = sendRequest("POST", "/products/{productName}/features/{featureName}", p1, productName, featureName);
        Assert.assertTrue(s2 < 300);

        int s3 = sendRequest("POST", "/products/{productName}/configurations/{configurationName}", null, productName, configName);
        Assert.assertTrue(s3 < 300);

        Assert.assertTrue(s4 < 300);

        int s5 = sendRequest("DELETE", "/products/{productName}/features/{featureName}", null, productName, featureName);
        Assert.assertEquals(204, s5);
    }
