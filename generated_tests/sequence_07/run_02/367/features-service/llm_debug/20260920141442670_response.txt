package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class ConstraintRequiresTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCreateRequiresConstraintReturns201() {
        String productName = "TestProduct-" + java.util.UUID.randomUUID();
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/FeatureA").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/FeatureB").then().statusCode(lessThan(300));

        given()
            .formParam("sourceFeature", "FeatureA")
            .formParam("requiredFeature", "FeatureB")
        .when()
            .post("/products/" + productName + "/constraints/requires")
        .then()
            .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetProductReturnsRequiresConstraintInfo() {
        String productName = "TestProduct-" + java.util.UUID.randomUUID();
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/SourceFeat").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/RequiredFeat").then().statusCode(lessThan(300));
        given()
            .formParam("sourceFeature", "SourceFeat")
            .formParam("requiredFeature", "RequiredFeat")
        .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));

        given()
        .when()
            .get("/products/" + productName)
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluateSourceActiveRequiredNotActive() {
        String productName = "TestProduct-" + java.util.UUID.randomUUID();
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/SourceA").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/RequiredA").then().statusCode(lessThan(300));
        given()
            .formParam("sourceFeature", "SourceA")
            .formParam("requiredFeature", "RequiredA")
        .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigA").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigA/features/SourceA").then().statusCode(lessThan(300));

        given()
        .when()
            .get("/products/" + productName + "/configurations/ConfigA/features")
        .then()
            .body("", hasItem("RequiredA"));
    }

    @Test(timeout = 60000)
    public void testEvaluateSourceNotActive() {
        String productName = "TestProduct-" + java.util.UUID.randomUUID();
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/SourceB").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/RequiredB").then().statusCode(lessThan(300));
        given()
            .formParam("sourceFeature", "SourceB")
            .formParam("requiredFeature", "RequiredB")
        .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigB").then().statusCode(lessThan(300));

        given()
        .when()
            .get("/products/" + productName + "/configurations/ConfigB/features")
        .then()
            .body("", not(hasItem("RequiredB")));
    }

    @Test(timeout = 60000)
    public void testEvaluateBothSourceAndRequiredActive() {
        String productName = "TestProduct-" + java.util.UUID.randomUUID();
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/SourceC").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/RequiredC").then().statusCode(lessThan(300));
        given()
            .formParam("sourceFeature", "SourceC")
            .formParam("requiredFeature", "RequiredC")
        .when().post("/products/" + productName + "/constraints/requires").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigC").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigC/features/SourceC").then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/ConfigC/features/RequiredC").then().statusCode(lessThan(300));

        given()
        .when()
            .get("/products/" + productName + "/configurations/ConfigC/features")
        .then()
            .statusCode(200);
    }
}
