package ts01glm_5_2_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class BessjTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void bessj_xZero_returnsZero() {
        given()
                .when()
                .get("/api/bessj/3/0.0")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_negativeXLargeMagnitude_coversBessj1NegativeBranch() {
        given()
                .when()
                .get("/api/bessj/3/-10.0")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_axGreaterThanN_coversUpwardRecursion() {
        given()
                .when()
                .get("/api/bessj/3/2.5")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_axLessThanOrEqualN_coversDownwardRecursion() {
        given()
                .when()
                .get("/api/bessj/5/2.5")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_nLessThan2_returns400() {
        given()
                .when()
                .get("/api/bessj/0/2.5")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void bessj_invalidNType_returns400() {
        given()
                .when()
                .get("/api/bessj/abc/2.5")
                .then()
                .statusCode(400);
    }
}
