package ts01glm_5_2_3;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class ContributionTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void contributeWithValidAmountAndTokenReturns202() {
        given()
            .contentType(ContentType.JSON)
            .body("{\"amount\":1000,\"token\":\"tok_visa\",\"currency\":\"USD\"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(202);
    }

    @Test(timeout = 60000)
    public void contributeWithMissingTokenReturns400() {
        given()
            .contentType(ContentType.JSON)
            .body("{\"amount\":1000,\"currency\":\"USD\"}")
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void contributeWithMissingAmountReturns400() {
        given()
            .contentType(ContentType.JSON)
            .body("{\"token\":\"tok_visa\",\"currency\":\"USD\"}")

            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)

            .post("/contribute")
        .then()

    public void contributeWithNullTokenReturns400() {

    public void contributeWithEmptyTokenReturns400() {

    public void contributeWithExtraFieldsReturns202() {
