package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class CalcTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCalcAddOperation() {
        given()
                .when()
                .get("/api/calc/add/15.5/4.5")
                .then()
                .statusCode(200)
                .body(containsString("20.0"));
    }

    @Test(timeout = 60000)
    public void testCalcDivideByZero() {
        given()
                .when()
                .get("/api/calc/divide/100/0")
                .then()
                .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testCalcInvalidNumberFormat() {
        given()
                .when()
                .get("/api/calc/add/10/twenty")
                .then()
                .statusCode(400);
    }
}
