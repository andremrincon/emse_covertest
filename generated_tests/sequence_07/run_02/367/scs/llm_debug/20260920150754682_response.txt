package ts01glm_5_2_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class FileSuffixTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv().getOrDefault("API_BASE_URL", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testTextDirectoryWithTxtSuffix() {
        given()
            .when()
                .get("/api/filesuffix/text/file.txt")
            .then()
                .statusCode(200)
                .body(containsString("1"));
    }

    @Test(timeout = 60000)
    public void testAcrobatDirectoryWithPdfSuffix() {
        given()
            .when()
                .get("/api/filesuffix/acrobat/file.pdf")
            .then()
                .statusCode(200)
                .body(containsString("2"));
    }

    @Test(timeout = 60000)
    public void testWordDirectoryWithDocSuffix() {
        given()
            .when()
                .get("/api/filesuffix/word/file.doc")
            .then()
                .statusCode(200)
                .body(containsString("3"));
    }

    @Test(timeout = 60000)
    public void testBinDirectoryWithExeSuffix() {
        given()
            .when()
                .get("/api/filesuffix/bin/file.exe")
            .then()
                .statusCode(200)
                .body(containsString("4"));
    }

    @Test(timeout = 60000)

        given()
            .when()
                .get("/api/filesuffix/lib/file.dll")
            .then()
                .statusCode(200)

    @Test(timeout = 60000)
    public void testNoDotInFileReturnsZero() {
        given()
            .when()
                .get("/api/filesuffix/text/nofile")
            .then()
                .statusCode(200)
                .body(containsString("0"));
    }
}
