package ts01glm_5_2_3;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class Ordered4Test {

    private static String baseUrl;

    @BeforeClass
    public static void setup() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testIncreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/apple/banana/zebra/cherry")
            .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testDecreasingOrder() {
        given()
            .when()
                .get("/api/ordered4/zzzzz/yyyyy/wwwww/xxxxx")
            .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenLengthPassesNeither() {
        given()
            .when()
                .get("/api/ordered4/apple/apple/apple/apple")
            .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenLengthTooShort() {
        given()
            .when()
                .get("/api/ordered4/a/b/c/d")
            .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testUnorderedWhenLengthTooLong() {
        given()
            .when()
                .get("/api/ordered4/aaaaaaa/bbbbbbb/ccccccc/ddddddd")
            .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testServerErrorWithSpaceY() {
        given()
            .pathParam("w", "zebra")
            .pathParam("x", "yak")
            .pathParam("z", "x-ray")
            .pathParam("y", " ")
            .when()
                .get("/api/ordered4/{w}/{x}/{z}/{y}")
            .then()
                .statusCode(500);
    }
}
