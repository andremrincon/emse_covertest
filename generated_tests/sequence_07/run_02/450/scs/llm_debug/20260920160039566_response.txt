package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class TitleTest {

    @Before
    public void setUp() {
        String baseURI = System.getenv("BASE_URI");
        if (baseURI == null || baseURI.isEmpty()) {
            baseURI = "http://localhost:8080";
        }
        RestAssured.baseURI = baseURI;
    }

    @Test(timeout = 60000)
    public void testMaleWithMatchingTitleMr() {
        given()
            .when()
                .get("/api/title/male/mr")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testMaleWithNonMatchingTitle() {
        given()
            .when()
                .get("/api/title/male/xyz")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFemaleWithMatchingTitleMrs() {
        given()
            .when()
                .get("/api/title/female/mrs")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFemaleWithNonMatchingTitle() {
        given()
            .when()
                .get("/api/title/female/xyz")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNoneWithMatchingTitleDr() {
        given()
            .when()
                .get("/api/title/none/dr")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnknownSexReturnsError() {
        given()
            .when()
                .get("/api/title/other/mr")
            .then()
                .statusCode(500);
    }
}
