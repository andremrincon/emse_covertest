package ts01glm_5_2_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

public class CurrencyTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGetCurrencyByCodeUSD() {
        given()
                .when()
                .get("/v1/currency/USD")
                .then()
                .statusCode(200)
                .body("[0].currencies[0].code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeUSReturnsCurrency() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies[0].name", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetV2CurrencyEUR() {
        given()
                .when()
                .get("/v2/currency/EUR")
                .then()
                .statusCode(200)
                .body("[0].currencies[0].symbol", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetV2RegionalblocEUWithCurrencyFields() {
        given()
                .queryParam("fields", "name;capital;currencies")
                .when()
                .get("/v2/regionalbloc/EU")
                .then()
                .statusCode(200)
                .body("[0].currencies[0].code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetV1CurrencyInvalidNumericReturns400() {
        given()
                .when()
                .get("/v1/currency/123")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetV1CurrencyNotFoundReturns404() {
        given()
                .when()
                .get("/v1/currency/XYZ")
                .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetV2AlphaUSWithCurrencyFields() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies[0].code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetV1AlphaCodesReturnsCurrency() {
        given()
                .queryParam("codes", "US,CA")
                .when()
                .get("/v1/alpha")
                .then()
                .statusCode(200)
                .body("[0].currencies[0].name", notNullValue());
    }

    @Test(timeout = 60000)
    public void testGetV2AllWithCurrencyFields() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/all")
                .then()
                .statusCode(200)
                .body("[0].currencies[0].symbol", notNullValue());
    }
}
