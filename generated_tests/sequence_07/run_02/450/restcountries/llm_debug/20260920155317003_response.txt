package ts01glm_5_2_3;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testV1CurrencyEndpointReturnsCountriesWithCurrencyCode() {
        given()
                .when()
                .get("/v1/currency/USD")
                .then()
                .statusCode(200)
                .body("currencies[0].code", hasItem("USD"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaEndpointReturnsCountryWithCurrencyName() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies[0].name", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaEndpointReturnsCountryWithCurrencySymbol() {
        given()
                .when()
                .get("/v1/alpha/GB")
                .then()
                .statusCode(200)
                .body("currencies[0].symbol", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCountriesWithCurrencyFields() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/currency/EUR")
                .then()
                .statusCode(200)
                .body("currencies[0][0].code", equalTo("EUR"));
    }

    @Test(timeout = 60000)
    public void testV2AlphaEndpointReturnsCountryWithCurrencyName() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/alpha/DE")
                .then()
                .statusCode(200)
                .body("currencies[0].name", equalTo("Euro"));
    }

    @Test(timeout = 60000)
    public void testV2AlphaEndpointReturnsCountryWithCurrencySymbol() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies[0].symbol", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AllReturnsCountriesWithCurrencyObjects() {
        given()
                .when()
                .get("/v1/all")
                .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2RegionalblocReturnsCountriesWithCurrencyCode() {
        given()
                .queryParam("fields", "name;currencies")
                .when()
                .get("/v2/regionalbloc/EU")
                .then()
                .statusCode(200)
                .body("currencies[0][0].code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesEndpointReturnsCountriesWithCurrencyFields() {
        given()
                .queryParam("codes", "US,CA")
                .when()
                .get("/v1/alpha")
                .then()
                .statusCode(200)
                .body("currencies[0][0].code", notNullValue());
    }
}
