public class CORSFilterTest {

    private static String baseUrl;

    @BeforeClass
    public static void setup() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
    }

    private static String buildPath(String template, Object... args) throws Exception {
        String result = template;
        Pattern p = Pattern.compile("\\{[^}]+\\}");
        Matcher m = p.matcher(result);
        int i = 0;
        while (m.find() && i < args.length) {
            String encoded = URLEncoder.encode(String.valueOf(args[i]), "UTF-8");
            result = result.substring(0, m.start()) + encoded + result.substring(m.end());
            m = p.matcher(result);
            i++;
        }
        return result;
    }

    private static int sendRequest(String method, String pathTemplate, String body, String contentType, Object... args) throws Exception {
        String path = buildPath(pathTemplate, args);
        URL url = new URL(baseUrl + path);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(30000);
        conn.setReadTimeout(30000);
        if (contentType != null) {
            conn.setRequestProperty("Content-Type", contentType);
        }
        if (body != null && ("POST".equals(method) || "PUT".equals(method) || "PATCH".equals(method))) {
            conn.setDoOutput(true);
            byte[] out = body.getBytes("UTF-8");
            conn.setRequestProperty("Content-Length", String.valueOf(out.length));
            OutputStream os = conn.getOutputStream();
            try {
                os.write(out);
                os.flush();
            } finally {
                try { os.close(); } catch (IOException ignore) {}
            }
        }
        int status;
        try {
            status = conn.getResponseCode();
            InputStream is = status >= 400 ? conn.getErrorStream() : conn.getInputStream();
            if (is != null) {
                try {
                    byte[] buf = new byte[1024];
                    while (is.read(buf) != -1) { }
                } finally {
                    try { is.close(); } catch (IOException ignore) {}
                }
            }
        } finally {
            conn.disconnect();
        }
        return status;
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigurationWithNoConstraints() throws Exception {
        String productName = "EvalTest-NoConstraints-" + UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "default-config";

        int status;

        status = sendRequest("POST", "/products/{productName}", null, null, productName);
        Assert.assertTrue(status < 300);

        status = sendRequest("POST", "/products/{productName}/configurations/{configurationName}", null, null, productName, configurationName);
        Assert.assertTrue(status < 300);

        status = sendRequest("GET", "/products/{productName}/configurations/{configurationName}/features", null, null, productName, configurationName);
        Assert.assertEquals(200, status);
    }
