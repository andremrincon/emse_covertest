    @BeforeClass

        baseUrl = System.getenv("BASE_URL");

    private static class SimpleResponse {
        final int statusCode;
        final String body;

        SimpleResponse(int statusCode, String body) {
            this.statusCode = statusCode;
            this.body = body;
        }
    }

    private String buildPath(String pathTemplate, String... params) throws Exception {
        String result = pathTemplate;
        for (String p : params) {
            String encoded = URLEncoder.encode(p, "UTF-8");
            result = result.replaceFirst("\\{[^/}]+\\}", encoded);
        }
        return result;
    }

    private SimpleResponse sendRequest(String method, String pathTemplate, String... params) {
        try {
            String path = buildPath(pathTemplate, params);
            String urlStr = baseUrl + path;
            URL url = new URL(urlStr);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod(method);
            conn.setConnectTimeout(60000);
            conn.setReadTimeout(60000);
            if ("POST".equals(method) || "PUT".equals(method)) {
                conn.setDoOutput(true);
                conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
                String body = "";
                byte[] out = body.getBytes("UTF-8");
                conn.setFixedLengthStreamingMode(out.length);
                try (OutputStream os = conn.getOutputStream()) {
                    os.write(out);
                }
            }
            int status = conn.getResponseCode();
            InputStream is;
            try {
                if (status >= 400) {
                    is = conn.getErrorStream();
                } else {
                    is = conn.getInputStream();
                }
            } catch (Exception e) {
                is = null;
            }
            StringBuilder sb = new StringBuilder();
            if (is != null) {
                try (BufferedReader br = new BufferedReader(new InputStreamReader(is, "UTF-8"))) {
                    String line;
                    while ((line = br.readLine()) != null) {
                        sb.append(line).append("\n");
                    }
                }
            }
            String respBody = sb.toString().trim();
            conn.disconnect();
            return new SimpleResponse(status, respBody);
        } catch (Exception e) {
            return new SimpleResponse(500, "");
        }
    }

    @Test(timeout = 60000)
    public void getConfigurationsNamesForProduct_withExistingConfigurations_returnsNames() {
        String productName = "test-prod-" + java.util.UUID.randomUUID().toString().substring(0, 8);
        String configName = "config-" + java.util.UUID.randomUUID().toString().substring(0, 8);

        SimpleResponse r1 = sendRequest("POST", "/products/{productName}", productName);
        assertTrue(r1.statusCode < 300);
        SimpleResponse r2 = sendRequest("POST", "/products/{productName}/configurations/{configurationName}", productName, configName);
        assertTrue(r2.statusCode < 300);

        SimpleResponse r3 = sendRequest("GET", "/products/{productName}/configurations", productName);
        assertEquals(200, r3.statusCode);
        assertTrue(r3.body.contains(configName));
    }
